📚 什麼是 JavaScript？

JavaScript 是網頁的程式語言，讓網頁能夠與使用者互動、動態更新內容、處理表單驗證等。

🖱️ 互動功能

點擊按鈕、表單驗證、彈出視窗

🔄 動態更新

不重新整理頁面就更新內容

🌐 API 通訊

從後端取得資料

🎨 動畫效果

建立流暢的視覺效果

JavaScript 加入網頁

<!-- 1. 內部腳本 --> <script> console.log('Hello World!'); </script> <!-- 2. 外部腳本（推薦！放在 body 結尾前）--> <script src="script.js"></script>

📦 變數宣告

// let - 可重新賦值的變數（推薦使用） let temperature = 25; temperature = 30; // 可以改變 // const - 常數，不可重新賦值 const PI = 3.14159; // PI = 3; // 錯誤！常數不能改變 // var - 舊式宣告（不建議使用） var oldStyle = '避免使用';

資料型態

型態 範例 說明
Number 42, 3.14, -10 數字（整數或小數）
String 'Hello', "World", `模板` 文字字串
Boolean true, false 布林值
Array [1, 2, 3] 陣列
Object {name: 'Tom', age: 20} 物件
null null 空值
undefined undefined 未定義
// 字串模板（推薦！） const temp = 25; const humidity = 60; const message = `目前溫度 ${temp}°C，濕度 ${humidity}%`; // 陣列 const crops = ['水稻', '玉米', '小麥']; console.log(crops[0]); // '水稻' console.log(crops.length); // 3 // 物件 const sensor = { type: '溫度感測器', value: 25.5, unit: '°C', isActive: true }; console.log(sensor.value); // 25.5 console.log(sensor['type']); // '溫度感測器'

🔢 運算子

// 算術運算子 let a = 10, b = 3; console.log(a + b); // 13 console.log(a - b); // 7 console.log(a * b); // 30 console.log(a / b); // 3.333... console.log(a % b); // 1 (餘數) console.log(a ** b); // 1000 (次方) // 比較運算子 console.log(5 == '5'); // true（只比較值） console.log(5 === '5'); // false（比較值和型態，推薦！） console.log(5 !== 3); // true console.log(5 > 3); // true console.log(5 >= 5); // true // 邏輯運算子 console.log(true && false); // false (且) console.log(true || false); // true (或) console.log(!true); // false (非)
⚠️ 請用 === 而非 ==
=== 會同時比較值和型態，更加安全。
例如：5 == '5' 是 true，但 5 === '5' 是 false

🔁 迴圈

const sensors = ['溫度', '濕度', '光照', '土壤']; // for 迴圈 for (let i = 0; i < sensors.length; i++) { console.log(sensors[i]); } // for...of（推薦！遍歷陣列元素） for (const sensor of sensors) { console.log(sensor); } // for...in（遍歷物件屬性） const data = { temp: 25, humidity: 60 }; for (const key in data) { console.log(`${key}: ${data[key]}`); } // forEach（陣列方法） sensors.forEach((sensor, index) => { console.log(`${index + 1}. ${sensor}`); }); // while 迴圈 let count = 0; while (count < 3) { console.log(count); count++; }

⚙️ 函式

// 函式宣告 function checkTemperature(temp) { if (temp > 35) { return '過熱'; } else if (temp > 20) { return '適中'; } else { return '偏冷'; } } console.log(checkTemperature(28)); // '適中' // 箭頭函式（推薦！更簡潔） const add = (a, b) => a + b; console.log(add(5, 3)); // 8 // 多行箭頭函式 const getSensorStatus = (value, threshold) => { if (value > threshold) { return { status: 'warning', message: '超過閾值' }; } return { status: 'normal', message: '正常' }; }; // 預設參數 const greet = (name = '農友') => `你好，${name}！`; console.log(greet()); // '你好，農友！'

🎮 DOM 操作

DOM (Document Object Model) 讓 JavaScript 可以操作 HTML 元素。

選取元素

// 用 ID 選取（回傳單一元素） const title = document.getElementById('main-title'); // 用 CSS 選擇器選取（回傳第一個符合的） const card = document.querySelector('.sensor-card'); // 用 CSS 選擇器選取所有符合的（回傳 NodeList） const allCards = document.querySelectorAll('.sensor-card'); // 遍歷所有元素 allCards.forEach(card => { console.log(card.textContent); });

修改元素

const tempDisplay = document.getElementById('temp-value'); // 修改文字內容 tempDisplay.textContent = '28°C'; // 修改 HTML 內容 tempDisplay.innerHTML = '<strong>28</strong>°C'; // 修改樣式 tempDisplay.style.color = '#ff0000'; tempDisplay.style.fontSize = '24px'; // 修改 class tempDisplay.classList.add('warning'); tempDisplay.classList.remove('normal'); tempDisplay.classList.toggle('active'); // 修改屬性 const img = document.querySelector('img'); img.setAttribute('src', 'new-image.jpg'); img.alt = '新圖片說明';

建立與移除元素

// 建立新元素 const newCard = document.createElement('div'); newCard.className = 'sensor-card'; newCard.innerHTML = ` <h3>新感測器</h3> <p>數值: 100</p> `; // 加入到頁面 const container = document.getElementById('sensor-container'); container.appendChild(newCard); // 移除元素 newCard.remove();

🌐 Fetch API（取得後端資料）

// GET 請求 fetch('/api/sensors') .then(response => response.json()) .then(data => { console.log('感測器資料:', data); }) .catch(error => { console.error('錯誤:', error); }); // 使用 async/await（推薦！更易讀） async function getSensorData() { try { const response = await fetch('/api/sensors'); const data = await response.json(); console.log(data); return data; } catch (error) { console.error('錯誤:', error); } } // POST 請求 async function sendCommand(command) { const response = await fetch('/api/control', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: command }) }); return response.json(); }

⏱️ 定時器

// setTimeout - 延遲執行一次 setTimeout(() => { console.log('3 秒後執行'); }, 3000); // 3000 毫秒 = 3 秒 // setInterval - 每隔一段時間重複執行 let intervalId = setInterval(() => { updateSensorDisplay(); }, 5000); // 每 5 秒更新一次 // 停止定時器 clearInterval(intervalId); // 實際應用：即時更新感測器數據 async function startRealTimeUpdate() { const updateDisplay = async () => { const data = await fetch('/api/sensors').then(r => r.json()); document.getElementById('temp').textContent = data.temperature + '°C'; document.getElementById('humidity').textContent = data.humidity + '%'; }; updateDisplay(); // 立即執行一次 setInterval(updateDisplay, 5000); // 然後每 5 秒更新 }

// 農場監控系統 const FarmMonitor = { // 設定閾值 thresholds: { tempHigh: 35, tempLow: 10, humidityLow: 30 }, // 初始化 init() { this.bindEvents(); this.startMonitoring(); }, // 綁定事件 bindEvents() { document.getElementById('irrigate-btn') .addEventListener('click', () => this.startIrrigation()); document.getElementById('fan-btn') .addEventListener('click', () => this.toggleFan()); }, // 開始監控 startMonitoring() { this.fetchAndUpdate(); setInterval(() => this.fetchAndUpdate(), 5000); }, // 取得並更新資料 async fetchAndUpdate() { try { const data = await fetch('/api/sensors').then(r => r.json()); this.updateDisplay(data); this.checkAlerts(data); } catch (err) { console.error('資料取得失敗:', err); } }, // 更新顯示 updateDisplay(data) { document.getElementById('temp-value').textContent = `${data.temp}°C`; document.getElementById('humidity-value').textContent = `${data.humidity}%`; }, // 檢查警報 checkAlerts(data) { const tempEl = document.getElementById('temp-value'); if (data.temp > this.thresholds.tempHigh) { tempEl.classList.add('danger'); this.showNotification('溫度過高！'); } else { tempEl.classList.remove('danger'); } }, // 啟動灌溉 async startIrrigation() { await fetch('/api/irrigate', { method: 'POST' }); this.showNotification('灌溉已啟動！'); }, // 顯示通知 showNotification(message) { const notification = document.createElement('div'); notification.className = 'notification'; notification.textContent = message; document.body.appendChild(notification); setTimeout(() => notification.remove(), 3000); } }; // 頁面載入後初始化 document.addEventListener('DOMContentLoaded', () => { FarmMonitor.init(); });
