關於序列埠通訊 (Serial Communication)

什麼是序列埠通訊？

序列埠通訊（Serial Communication） 是一種資料傳輸方式，資料以「一個位元接著一個位元」的順序傳送，就像排隊過獨木橋一樣，一次只能過一個人。

序列傳輸 vs 並列傳輸：

序列傳輸（Serial）：
┌─────┐    ─→─→─→─→─→─→─→─→    ┌─────┐
│ 發送 │    1 0 1 1 0 1 0 0    │ 接收 │
│  端  │    ← 一條線傳輸 →      │  端  │
└─────┘                        └─────┘
        資料一個一個排隊傳送

並列傳輸（Parallel）：
┌─────┐    ══════════════════    ┌─────┐
│ 發送 │    1 ─────────→         │ 接收 │
│  端  │    0 ─────────→         │  端  │
│     │    1 ─────────→         │     │
│     │    1 ─────────→ 8條線    │     │
│     │    0 ─────────→         │     │
│     │    1 ─────────→         │     │
│     │    0 ─────────→         │     │
│     │    0 ─────────→         │     │
└─────┘                          └─────┘
        資料同時傳送（較快但線多）

為什麼選擇序列通訊？

特點 序列通訊 並列通訊
線材數量 少（1-2條） 多（8條以上）
傳輸距離 遠 近
成本 低 高
複雜度 簡單 複雜
速度 較慢 較快

現代裝置大多使用序列通訊（USB、I2C、SPI、UART 都是序列通訊）！

UART 序列通訊

Arduino 最常用的序列通訊方式是 UART（Universal Asynchronous Receiver/Transmitter），也就是我們常說的「序列埠」。

UART 的特點

UART 通訊特性：
┌────────────────────────────────────┐
│ • 非同步通訊（不需要時鐘線）        │
│ • 全雙工（可同時收發）              │
│ • 點對點通訊（一對一）              │
│ • 需要約定相同的鮑率（Baud Rate）   │
└────────────────────────────────────┘

UART 接線

兩個裝置的 UART 連接：

裝置 A              裝置 B
┌──────┐            ┌──────┐
│      │            │      │
│   TX ├──────────→─┤ RX   │  （A 發送 → B 接收）
│      │            │      │
│   RX ├─←──────────┤ TX   │  （A 接收 ← B 發送）
│      │            │      │
│  GND ├────────────┤ GND  │  （共同接地）
│      │            │      │
└──────┘            └──────┘

重點：TX 接 RX，RX 接 TX（交叉連接）！

鮑率（Baud Rate）

鮑率 = 每秒傳送的符號數（近似於每秒位元數）

常見鮑率：
┌──────────────────────────────────┐
│   9600  ← 最常用，穩定           │
│  19200                           │
│  38400                           │
│  57600                           │
│ 115200  ← 較快，常用於除錯       │
│ 250000                           │
│ 500000                           │
│ 1000000 ← 高速，需要好的線材     │
└──────────────────────────────────┘

⚠️ 重要：發送端和接收端必須使用相同的鮑率！

UART 資料格式

一個位元組的傳輸格式：

空閒 ─┐ ┌─起始位元─┬─資料位元（8位）─┬─停止位元─┐ ┌─ 空閒
      │ │         │                │         │ │
  ────┘ └─────────┴────────────────┴─────────┘ └────

     HIGH  LOW    D0 D1 D2 D3 D4 D5 D6 D7    HIGH

常見設定：8N1
• 8：8 個資料位元
• N：無同位檢查（No Parity）
• 1：1 個停止位元

Arduino 的 Serial 函式庫

Arduino 內建 Serial 函式庫，讓序列通訊變得非常簡單。

基本函式一覽

// 初始化
Serial.begin(9600);        // 設定鮑率

// 發送資料
Serial.print("Hello");     // 發送文字（不換行）
Serial.println("World");   // 發送文字（換行）
Serial.write(65);          // 發送原始位元組（'A'）

// 接收資料
Serial.available();        // 檢查緩衝區有多少資料
Serial.read();             // 讀取一個位元組
Serial.readString();       // 讀取字串
Serial.parseInt();         // 讀取整數

// 其他
Serial.end();              // 結束序列通訊
Serial.flush();            // 等待發送完成

程式範例

範例 1：基本輸出（Hello World）

/*
 * 序列埠基本輸出
 * 開啟序列埠監視器觀看結果
 */

void setup() {
  // 初始化序列埠，鮑率 9600
  Serial.begin(9600);

  // 等待序列埠連接（某些板子需要）
  while (!Serial) {
    ; // 等待
  }

  Serial.println("=== Arduino 序列埠測試 ===");
  Serial.println();
}

void loop() {
  Serial.println("Hello, World!");
  delay(1000);
}

範例 2：print vs println vs write

/*
 * 三種輸出方式的差異
 */

void setup() {
  Serial.begin(9600);
  Serial.println("=== 輸出方式比較 ===");
  Serial.println();

  // print：輸出文字，不換行
  Serial.print("AAA");
  Serial.print("BBB");
  Serial.print("CCC");
  Serial.println();  // 手動換行

  // println：輸出文字，自動換行
  Serial.println("AAA");
  Serial.println("BBB");
  Serial.println("CCC");

  // 數字輸出
  int num = 65;
  Serial.print("print(65) = ");
  Serial.println(num);        // 輸出 "65"

  Serial.print("write(65) = ");
  Serial.write(num);          // 輸出 "A"（ASCII 碼 65）
  Serial.println();

  // 不同進位顯示
  Serial.print("255 的不同表示：");
  Serial.println();
  Serial.print("  十進位: ");
  Serial.println(255);        // 255
  Serial.print("  二進位: ");
  Serial.println(255, BIN);   // 11111111
  Serial.print("  十六進位: ");
  Serial.println(255, HEX);   // FF
  Serial.print("  八進位: ");
  Serial.println(255, OCT);   // 377
}

void loop() {
  // 空的
}

範例 3：讀取使用者輸入

/*
 * 序列埠讀取使用者輸入
 * 在序列埠監視器輸入文字，Arduino 會回應
 */

void setup() {
  Serial.begin(9600);
  Serial.println("=== 序列埠輸入測試 ===");
  Serial.println("請在上方輸入文字，然後按 Enter");
  Serial.println();
}

void loop() {
  // 檢查是否有資料可讀
  if (Serial.available() > 0) {
    // 讀取整行文字
    String input = Serial.readStringUntil('\n');
    input.trim();  // 移除前後空白

    // 回應
    Serial.print("你輸入了: ");
    Serial.println(input);

    // 顯示字數
    Serial.print("字元數: ");
    Serial.println(input.length());
    Serial.println();
  }
}

範例 4：指令控制 LED

/*
 * 用序列埠指令控制 LED
 * 輸入 "on" 開燈，輸入 "off" 關燈
 */

#define LED_PIN 13

void setup() {
  Serial.begin(9600);
  pinMode(LED_PIN, OUTPUT);

  Serial.println("=== LED 控制系統 ===");
  Serial.println("指令：");
  Serial.println("  on  - 開燈");
  Serial.println("  off - 關燈");
  Serial.println();
}

void loop() {
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    cmd.toLowerCase();  // 轉小寫

    if (cmd == "on") {
      digitalWrite(LED_PIN, HIGH);
      Serial.println("LED 已開啟");
    }
    else if (cmd == "off") {
      digitalWrite(LED_PIN, LOW);
      Serial.println("LED 已關閉");
    }
    else {
      Serial.print("未知指令: ");
      Serial.println(cmd);
    }
  }
}

範例 5：讀取數字與計算

/*
 * 讀取數字並進行計算
 */

void setup() {
  Serial.begin(9600);
  Serial.println("=== 簡易計算機 ===");
  Serial.println("輸入一個數字：");
}

void loop() {
  if (Serial.available() > 0) {
    // 讀取整數
    int num = Serial.parseInt();

    // parseInt 會讀到非數字為止
    // 清除剩餘的換行符
    while (Serial.available()) {
      Serial.read();
    }

    if (num != 0) {
      Serial.print("你輸入: ");
      Serial.println(num);

      Serial.print("平方: ");
      Serial.println((long)num * num);

      Serial.print("兩倍: ");
      Serial.println(num * 2);

      Serial.print("一半: ");
      Serial.println(num / 2.0);

      Serial.println();
      Serial.println("輸入下一個數字：");
    }
  }
}

範例 6：JSON 格式資料傳輸

/*
 * 發送 JSON 格式的感測器資料
 * 常用於與電腦程式或網頁溝通
 */

void setup() {
  Serial.begin(9600);
  Serial.println("=== JSON 資料發送 ===");
}

void loop() {
  // 模擬感測器資料
  float temperature = 25.0 + random(-50, 50) / 10.0;
  float humidity = 65.0 + random(-100, 100) / 10.0;
  int lightLevel = random(300, 800);

  // 組合 JSON 字串
  Serial.print("{");
  Serial.print("\"temperature\":");
  Serial.print(temperature, 1);
  Serial.print(",\"humidity\":");
  Serial.print(humidity, 1);
  Serial.print(",\"light\":");
  Serial.print(lightLevel);
  Serial.println("}");

  delay(2000);
}

/*
 * 輸出範例：
 * {"temperature":25.3,"humidity":64.2,"light":523}
 * {"temperature":24.8,"humidity":66.1,"light":487}
 */

Arduino 與電腦溝通

使用 Python 讀取 Arduino 資料

# Python 程式：serial_reader.py
# 需要安裝：pip install pyserial

import serial
import time

# 開啟序列埠（請修改 COM 埠號）
ser = serial.Serial('COM3', 9600, timeout=1)
time.sleep(2)  # 等待 Arduino 重置

print("開始接收 Arduino 資料...")

try:
    while True:
        if ser.in_waiting > 0:
            line = ser.readline().decode('utf-8').strip()
            print(f"收到: {line}")
except KeyboardInterrupt:
    print("結束程式")
finally:
    ser.close()

使用 Python 發送指令給 Arduino

# Python 程式：serial_sender.py

import serial
import time

ser = serial.Serial('COM3', 9600, timeout=1)
time.sleep(2)

print("輸入指令（on/off），輸入 quit 離開")

while True:
    cmd = input("指令: ")
    if cmd.lower() == 'quit':
        break

    # 發送指令（加上換行符）
    ser.write((cmd + '\n').encode())

    # 等待並讀取回應
    time.sleep(0.1)
    while ser.in_waiting > 0:
        response = ser.readline().decode('utf-8').strip()
        print(f"Arduino 回應: {response}")

ser.close()

使用 Python 解析 JSON

# Python 程式：json_reader.py

import serial
import json
import time

ser = serial.Serial('COM3', 9600, timeout=1)
time.sleep(2)

print("接收 Arduino JSON 資料...")

try:
    while True:
        if ser.in_waiting > 0:
            line = ser.readline().decode('utf-8').strip()
            try:
                data = json.loads(line)
                print(f"溫度: {data['temperature']}°C")
                print(f"濕度: {data['humidity']}%")
                print(f"光線: {data['light']}")
                print("---")
            except json.JSONDecodeError:
                print(f"非 JSON 資料: {line}")
except KeyboardInterrupt:
    ser.close()

兩個 Arduino 之間的通訊

硬體連接

Arduino A (發送端)        Arduino B (接收端)
┌──────────────┐          ┌──────────────┐
│              │          │              │
│          TX1 ├────────→─┤ RX0          │
│              │          │              │
│          RX1 ├─←────────┤ TX0          │
│              │          │              │
│          GND ├──────────┤ GND          │
│              │          │              │
└──────────────┘          └──────────────┘

⚠️ 注意：
1. TX 接 RX，RX 接 TX（交叉）
2. GND 必須連接（共地）
3. 如果兩個 Arduino 電壓不同，需要電平轉換

發送端程式（Arduino A）

/*
 * Arduino A - 發送端
 * 每秒發送感測器資料
 */

void setup() {
  Serial.begin(9600);  // 用於與電腦溝通（除錯）
  Serial1.begin(9600); // 用於與另一個 Arduino 溝通
                       // Arduino UNO 只有一個硬體序列埠
                       // 需使用 SoftwareSerial

  Serial.println("發送端啟動");
}

void loop() {
  int sensorValue = analogRead(A0);

  // 發送給另一個 Arduino
  Serial1.print("SENSOR:");
  Serial1.println(sensorValue);

  // 同時顯示在電腦（除錯用）
  Serial.print("發送: ");
  Serial.println(sensorValue);

  delay(1000);
}

接收端程式（Arduino B）

/*
 * Arduino B - 接收端
 * 接收並解析資料
 */

void setup() {
  Serial.begin(9600);  // 與電腦溝通
  Serial1.begin(9600); // 與另一個 Arduino 溝通

  Serial.println("接收端啟動");
}

void loop() {
  if (Serial1.available() > 0) {
    String data = Serial1.readStringUntil('\n');
    data.trim();

    Serial.print("收到: ");
    Serial.println(data);

    // 解析資料
    if (data.startsWith("SENSOR:")) {
      int value = data.substring(7).toInt();
      Serial.print("感測器數值: ");
      Serial.println(value);
    }
  }
}

Arduino UNO 使用 SoftwareSerial

Arduino UNO 只有一個硬體序列埠（被 USB 佔用），需要用 SoftwareSerial：

/*
 * 使用 SoftwareSerial 建立第二個序列埠
 */

#include <SoftwareSerial.h>

// 定義軟體序列埠腳位（RX, TX）
SoftwareSerial mySerial(10, 11);  // RX=D10, TX=D11

void setup() {
  Serial.begin(9600);     // 硬體序列埠（USB）
  mySerial.begin(9600);   // 軟體序列埠

  Serial.println("SoftwareSerial 測試");
}

void loop() {
  // 從軟體序列埠讀取，發送到硬體序列埠
  if (mySerial.available()) {
    char c = mySerial.read();
    Serial.write(c);
  }

  // 從硬體序列埠讀取，發送到軟體序列埠
  if (Serial.available()) {
    char c = Serial.read();
    mySerial.write(c);
  }
}

序列埠監視器使用技巧

開啟方式

Arduino IDE：
• 工具 → 序列埠監視器
• 快捷鍵：Ctrl + Shift + M

設定：
• 鮑率：要與程式中 Serial.begin() 一致
• 行尾：選擇「換行」或「NL & CR」

行尾設定差異

行尾設定影響發送的內容：

「無行尾」：
  輸入 "hello" → 發送 "hello"

「換行」(Newline, NL)：
  輸入 "hello" → 發送 "hello\n"

「回車」(Carriage Return, CR)：
  輸入 "hello" → 發送 "hello\r"

「NL & CR」：
  輸入 "hello" → 發送 "hello\r\n"

建議：使用「換行」或「NL & CR」

序列繪圖家

Arduino IDE 內建的圖形化工具：
• 工具 → 序列繪圖家
• 快捷鍵：Ctrl + Shift + L

使用方式：
Serial.println(sensorValue);  // 單一數值

多條線：
Serial.print(value1);
Serial.print(",");
Serial.println(value2);  // 用逗號分隔

常見問題與解決

Q1：序列埠監視器顯示亂碼

原因：鮑率不匹配

解決：
1. 檢查程式中的 Serial.begin(鮑率)
2. 序列埠監視器選擇相同的鮑率

常見錯誤：
程式寫 Serial.begin(115200)
監視器選 9600
→ 顯示亂碼

Q2：找不到 COM 埠

Windows：
1. 開啟裝置管理員
2. 展開「連接埠 (COM 和 LPT)」
3. 找到 Arduino 對應的 COM 埠

如果看不到：
• 檢查 USB 線是否為數據線（非充電線）
• 安裝 Arduino 驅動程式
• 嘗試不同 USB 埠

Q3：上傳程式時顯示埠被佔用

原因：序列埠監視器佔用了埠

解決：
1. 關閉序列埠監視器
2. 上傳程式
3. 重新開啟監視器

或：Arduino IDE 2.0 會自動處理

Q4：Serial.read() 讀到 -1

// 錯誤寫法：
char c = Serial.read();  // 沒資料時回傳 -1

// 正確寫法：
if (Serial.available() > 0) {
  char c = Serial.read();  // 確定有資料才讀取
}

Q5：readString() 很慢

// readString() 會等待 timeout（預設 1 秒）

// 解決方法 1：設定較短的 timeout
Serial.setTimeout(100);  // 100 毫秒

// 解決方法 2：使用 readStringUntil()
String s = Serial.readStringUntil('\n');  // 讀到換行就停

序列通訊協定設計

簡單協定範例

設計一個控制 LED 的協定：

指令格式：<命令><參數>\n

範例：
  "L1\n"    → LED 開
  "L0\n"    → LED 關
  "B128\n"  → 亮度設為 128
  "S\n"     → 查詢狀態

實作程式

#define LED_PIN 9

int brightness = 0;
bool ledOn = false;

void setup() {
  Serial.begin(9600);
  pinMode(LED_PIN, OUTPUT);
  Serial.println("LED 控制系統就緒");
  Serial.println("指令: L1(開) L0(關) B0-255(亮度) S(狀態)");
}

void loop() {
  if (Serial.available() > 0) {
    char cmd = Serial.read();

    switch (cmd) {
      case 'L':
      case 'l':
        handleLED();
        break;

      case 'B':
      case 'b':
        handleBrightness();
        break;

      case 'S':
      case 's':
        handleStatus();
        break;

      case '\n':
      case '\r':
        // 忽略換行符
        break;

      default:
        Serial.print("未知指令: ");
        Serial.println(cmd);
    }
  }
}

void handleLED() {
  while (!Serial.available());  // 等待參數
  char param = Serial.read();

  if (param == '1') {
    ledOn = true;
    analogWrite(LED_PIN, brightness > 0 ? brightness : 255);
    Serial.println("OK: LED ON");
  } else if (param == '0') {
    ledOn = false;
    analogWrite(LED_PIN, 0);
    Serial.println("OK: LED OFF");
  }
}

void handleBrightness() {
  brightness = Serial.parseInt();
  brightness = constrain(brightness, 0, 255);

  if (ledOn) {
    analogWrite(LED_PIN, brightness);
  }

  Serial.print("OK: Brightness = ");
  Serial.println(brightness);
}

void handleStatus() {
  Serial.print("LED: ");
  Serial.print(ledOn ? "ON" : "OFF");
  Serial.print(", Brightness: ");
  Serial.println(brightness);
}
