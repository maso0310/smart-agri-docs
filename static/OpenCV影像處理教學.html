OpenCV 影像處理教學

OpenCV（Open Source Computer Vision Library）是最廣泛使用的電腦視覺庫，支援影像處理、物件偵測、機器學習等功能。

1. 安裝與基礎設定

1.1 安裝 OpenCV

# 使用 pip 安裝
pip install opencv-python

# 如果需要額外功能（如 SIFT）
pip install opencv-contrib-python

1.2 引入套件

4. 色彩空間轉換

4.1 常用轉換

# BGR 轉灰階
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# BGR 轉 RGB
rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

# BGR 轉 HSV
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# BGR 轉 LAB
lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)

4.2 HSV 色彩空間

通道 名稱 範圍 說明
H色相（Hue）0-179顏色種類
S飽和度（Saturation）0-255顏色鮮豔程度
V明度（Value）0-255明暗程度
# HSV 顏色過濾（以綠色為例）
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# 定義綠色範圍
lower_green = np.array([35, 50, 50])
upper_green = np.array([85, 255, 255])

# 建立遮罩
mask = cv2.inRange(hsv, lower_green, upper_green)

# 套用遮罩
result = cv2.bitwise_and(img, img, mask=mask)

5. 影像處理操作

5.1 調整尺寸

# 指定尺寸
resized = cv2.resize(img, (640, 480))

# 依比例縮放
scale = 0.5
resized = cv2.resize(img, None, fx=scale, fy=scale)

# 指定插值方法
resized = cv2.resize(img, (640, 480), interpolation=cv2.INTER_LINEAR)
# INTER_NEAREST：最近鄰插值（最快）
# INTER_LINEAR：雙線性插值（預設）
# INTER_CUBIC：雙三次插值（較慢但品質好）
# INTER_AREA：區域插值（縮小時推薦）

5.2 裁切與填充

# 裁切（使用 NumPy 切片）
cropped = img[100:300, 200:500]  # [y1:y2, x1:x2]

    img,
    top=10, bottom=10, left=10, right=10,
    borderType=cv2.BORDER_CONSTANT,

5.3 旋轉與翻轉

# 翻轉
flipped_h = cv2.flip(img, 1)   # 水平翻轉
flipped_v = cv2.flip(img, 0)   # 垂直翻轉
flipped_b = cv2.flip(img, -1)  # 同時翻轉

# 旋轉
height, width = img.shape[:2]
center = (width // 2, height // 2)

# 取得旋轉矩陣
rotation_matrix = cv2.getRotationMatrix2D(center, angle=45, scale=1.0)

# 執行旋轉
rotated = cv2.warpAffine(img, rotation_matrix, (width, height))

7. 二值化與閾值處理

7.1 全域閾值

9. 形態學操作

# 建立結構元素
kernel = np.ones((5, 5), np.uint8)

# 侵蝕（使白色區域縮小）
eroded = cv2.erode(binary, kernel, iterations=1)

# 膨脹（使白色區域擴大）
dilated = cv2.dilate(binary, kernel, iterations=1)

# 開運算（先侵蝕後膨脹，去除雜訊）
opened = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)

# 閉運算（先膨脹後侵蝕，填補空洞）
closed = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)

# 梯度（膨脹 - 侵蝕，取得邊緣）
gradient = cv2.morphologyEx(binary, cv2.MORPH_GRADIENT, kernel)

10. 輪廓檢測

# 尋找輪廓
contours, hierarchy = cv2.findContours(
    binary,
    cv2.RETR_EXTERNAL,      # 只取外輪廓
    cv2.CHAIN_APPROX_SIMPLE # 壓縮輪廓點
)

# 繪製輪廓
result = img.copy()
cv2.drawContours(result, contours, -1, (0, 255, 0), 2)

# 輪廓屬性
for cnt in contours:
    # 面積
    area = cv2.contourArea(cnt)

    # 周長
    perimeter = cv2.arcLength(cnt, True)

    # 外接矩形
    x, y, w, h = cv2.boundingRect(cnt)
    cv2.rectangle(result, (x, y), (x+w, y+h), (255, 0, 0), 2)

    # 外接圓
    (cx, cy), radius = cv2.minEnclosingCircle(cnt)
    cv2.circle(result, (int(cx), int(cy)), int(radius), (0, 0, 255), 2)
