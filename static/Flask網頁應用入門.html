Flask 網頁應用入門

使用 Python 建立網頁應用程式

📚 什麼是 Flask？

Flask 是一個輕量級的 Python 網頁框架，適合快速開發小型到中型的網頁應用。它簡單易學，非常適合作為後端 API 伺服器。

安裝 Flask

# 使用 pip 安裝 pip install flask

🚀 第一個 Flask 應用

from flask import Flask # 建立 Flask 應用 app = Flask(__name__) # 定義路由（首頁） @app.route('/') def home(): return '歡迎來到智慧農業系統！' # 另一個路由 @app.route('/about') def about(): return '這是關於頁面' # 啟動伺服器 if __name__ == '__main__': app.run(debug=True, port=5000)
💡 執行方式：
1. 儲存為 app.py
2. 執行 python app.py
3. 開啟瀏覽器訪問 http://localhost:5000

📡 建立 REST API
from flask import Flask, jsonify, request app = Flask(__name__) # 模擬資料 sensors = [ {'id': 1, 'name': '溫度', 'value': 25.5, 'unit': '°C'}, {'id': 2, 'name': '濕度', 'value': 60, 'unit': '%'} ] # GET - 取得所有感測器 @app.route('/api/sensors', methods=['GET']) def get_sensors(): return jsonify(sensors) # GET - 取得單一感測器 @app.route('/api/sensors/<int:sensor_id>', methods=['GET']) def get_sensor(sensor_id): sensor = next((s for s in sensors if s['id'] == sensor_id), None) if sensor: return jsonify(sensor) return jsonify({'error': '找不到感測器'}), 404 # POST - 執行控制命令 @app.route('/api/control', methods=['POST']) def control_action(): data = request.get_json() action = data.get('action') if action == 'irrigate': # 執行灌溉邏輯 return jsonify({'status': 'success', 'message': '灌溉已啟動'}) elif action == 'fan_on': return jsonify({'status': 'success', 'message': '風扇已開啟'}) return jsonify({'error': '未知動作'}), 400
🌾 農業監控系統完整範例
from flask import Flask, render_template, jsonify, request import random from datetime import datetime app = Flask(__name__) # 模擬感測器資料 def get_sensor_data(): return { 'temperature': round(random.uniform(20, 35), 1), 'humidity': round(random.uniform(40, 80), 1), 'soil_moisture': round(random.uniform(30, 70), 1), 'light': round(random.uniform(200, 1000), 0), 'timestamp': datetime.now().strftime('%Y-%m-%d %H:%M:%S') } # 首頁 @app.route('/') def index(): return render_template('index.html') # 儀表板 @app.route('/dashboard') def dashboard(): data = get_sensor_data() return render_template('dashboard.html', **data) # API: 取得即時感測器資料 @app.route('/api/sensors') def api_sensors(): return jsonify(get_sensor_data()) # API: 控制設備 @app.route('/api/control', methods=['POST']) def api_control(): data = request.get_json() device = data.get('device') action = data.get('action') # 實際應用中這裡會控制硬體 print(f'控制 {device}: {action}') return jsonify({ 'success': True, 'message': f'{device} 已{action}' }) if __name__ == '__main__': app.run(debug=True, host='0.0.0.0', port=5000)
