建立農業 GIS 視覺化應用

📚 什麼是 Leaflet？

Leaflet 是一個輕量級的 JavaScript 地圖函式庫，可以輕鬆建立互動式網頁地圖，適合展示農場位置、感測器分布、作物分區等地理資訊。

引入 Leaflet

<!-- 在 HTML head 中加入 --> <link rel="stylesheet" href="https://unpkg.com/leaflet@1.9.4/dist/leaflet.css"> <script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script> <!-- 地圖容器 --> <div id="map" style="height: 500px;"></div>

🌍 建立基本地圖

// 建立地圖，中心點設在台灣 const map = L.map('map').setView([23.5, 121], 8); // 加入圖磚圖層（底圖） L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '© OpenStreetMap', maxZoom: 19 }).addTo(map); // 其他常用圖磚來源 // 衛星影像: 'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}' // 地形圖: 'https://{s}.tile.opentopomap.org/{z}/{x}/{y}.png'

📍 標記與彈出視窗

// 基本標記 const marker = L.marker([25.033, 121.565]).addTo(map); // 加入彈出視窗 marker.bindPopup('<b>農場 A</b><br>溫度: 25°C'); // 自訂圖示 const sensorIcon = L.icon({ iconUrl: 'sensor-icon.png', iconSize: [32, 32], iconAnchor: [16, 32], popupAnchor: [0, -32] }); L.marker([25.0, 121.5], { icon: sensorIcon }) .bindPopup('溫度感測器') .addTo(map); // 批次加入多個標記 const sensors = [ { lat: 25.0, lng: 121.5, name: '感測器 A', temp: 25 }, { lat: 25.1, lng: 121.6, name: '感測器 B', temp: 28 }, { lat: 24.9, lng: 121.4, name: '感測器 C', temp: 22 } ]; sensors.forEach(sensor => { L.marker([sensor.lat, sensor.lng]) .bindPopup(`<b>${sensor.name}</b><br>溫度: ${sensor.temp}°C`) .addTo(map); });
🔷 繪製形狀
// 圓形（標示範圍） L.circle([25.0, 121.5], { color: 'green', fillColor: '#0f0', fillOpacity: 0.3, radius: 500 // 半徑（公尺） }).addTo(map); // 多邊形（農田邊界） const fieldBoundary = [ [25.00, 121.50], [25.01, 121.50], [25.01, 121.51], [25.00, 121.51] ]; L.polygon(fieldBoundary, { color: '#228b22', fillColor: '#90ee90', fillOpacity: 0.5 }).bindPopup('水稻田 A 區').addTo(map); // 線條（路徑、水道） const irrigationPath = [ [25.00, 121.50], [25.005, 121.505], [25.01, 121.51] ]; L.polyline(irrigationPath, { color: 'blue', weight: 3, dashArray: '5, 10' // 虛線 }).bindPopup('灌溉水道').addTo(map);
📊 GeoJSON 資料
// GeoJSON 格式的農田資料 const farmData = { "type": "FeatureCollection", "features": [ { "type": "Feature", "properties": { "name": "A區水稻田", "crop": "水稻", "ndvi": 0.72 }, "geometry": { "type": "Polygon", "coordinates": [[[121.50, 25.00], [121.51, 25.00], [121.51, 25.01], [121.50, 25.01], [121.50, 25.00]]] } } ] }; // 載入 GeoJSON 並根據屬性設定樣式 L.geoJSON(farmData, { style: function(feature) { const ndvi = feature.properties.ndvi; return { fillColor: ndvi > 0.6 ? '#228b22' : ndvi > 0.3 ? '#ffd700' : '#ff6347', weight: 2, color: 'white', fillOpacity: 0.7 }; }, onEachFeature: function(feature, layer) { layer.bindPopup(` <b>${feature.properties.name}</b><br> 作物: ${feature.properties.crop}<br> NDVI: ${feature.properties.ndvi} `); } }).addTo(map);
🖱️ 互動事件
// 地圖點擊事件 map.on('click', function(e) { const { lat, lng } = e.latlng; console.log(`點擊座標: ${lat}, ${lng}`); // 在點擊位置加入標記 L.marker([lat, lng]) .bindPopup(`新增感測器<br>座標: ${lat.toFixed(4)}, ${lng.toFixed(4)}`) .addTo(map) .openPopup(); }); // 標記點擊事件 marker.on('click', function(e) { fetchSensorData(e.target.options.sensorId); }); // 縮放結束事件 map.on('zoomend', function() { console.log('目前縮放等級:', map.getZoom()); });
🌾 農業地圖完整範例
// 農場監控地圖 const FarmMap = { map: null, markers: [], init() { this.map = L.map('map').setView([23.5, 120.5], 12); L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', { attribution: 'Tiles © Esri' }).addTo(this.map); this.loadSensors(); this.loadFields(); }, async loadSensors() { const sensors = await fetch('/api/sensors').then(r => r.json()); sensors.forEach(sensor => { const color = sensor.temp > 30 ? 'red' : sensor.temp > 20 ? 'orange' : 'green'; const icon = L.divIcon({ className: 'sensor-marker', html: `<div style="background:${color}; padding:8px; border-radius:50%;">${sensor.temp}°</div>` }); L.marker([sensor.lat, sensor.lng], { icon }) .bindPopup(` <b>${sensor.name}</b><br> 溫度: ${sensor.temp}°C<br> 濕度: ${sensor.humidity}%<br> 更新: ${sensor.lastUpdate} `) .addTo(this.map); }); }, async loadFields() { const fields = await fetch('/api/fields').then(r => r.json()); L.geoJSON(fields, { style: feature => ({ fillColor: this.getNDVIColor(feature.properties.ndvi), weight: 2, color: 'white', fillOpacity: 0.6 }), onEachFeature: (feature, layer) => { layer.bindPopup(` <b>${feature.properties.name}</b><br> 作物: ${feature.properties.crop}<br> NDVI: ${feature.properties.ndvi}<br> 面積: ${feature.properties.area} 公頃 `); } }).addTo(this.map); }, getNDVIColor(ndvi) { return ndvi > 0.7 ? '#006400' : ndvi > 0.5 ? '#228b22' : ndvi > 0.3 ? '#ffd700' : '#ff6347'; } }; document.addEventListener('DOMContentLoaded', () => FarmMap.init());
