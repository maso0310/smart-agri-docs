ESP-01 是什麼？

ESP-01 是基於 ESP8266 晶片的 WiFi 模組，體積小巧、價格便宜，非常適合為 Arduino 專案添加網路功能。

重要警告：電壓問題

╔══════════════════════════════════════════════════════════════════╗
║  ⚠️ 重要警告：ESP-01 使用 3.3V 電壓！                           ║
╠══════════════════════════════════════════════════════════════════╣
║                                                                  ║
║  1. 電源必須是 3.3V                                              ║
║     - 直接接 5V 會燒毀 ESP-01！                                  ║
║     - Arduino Uno 的 3.3V 腳位電流可能不足                       ║
║     - 建議使用 AMS1117-3.3 穩壓模組                              ║
║                                                                  ║
║  2. 訊號電平也是 3.3V                                            ║
║     - Arduino TX (5V) 接 ESP-01 RX (3.3V) 需要分壓               ║
║     - ESP-01 TX (3.3V) 接 Arduino RX (5V) 可直接連接             ║
║                                                                  ║
║  3. 電流需求                                                     ║
║     - 平均 80mA，峰值可達 300mA                                  ║
║     - Arduino 3.3V 腳位只能提供約 50mA                           ║
║                                                                  ║
╚══════════════════════════════════════════════════════════════════╝

硬體接線

方案一：使用分壓電阻（基礎方案）

電源部分（使用 AMS1117-3.3 穩壓）：

    Arduino 5V ────────┬──────────────────────┐
                       │                      │
                  ┌────┴────┐                 │
                  │ AMS1117 │                 │
                  │  3.3V   │                 │

                       │                      │

                       │                      │
    Arduino GND ───────┴──── ESP-01 GND ──────┘

訊號部分（TX 需要分壓，RX 可直連）：

    Arduino                          ESP-01
    ┌─────────┐                    ┌─────────┐
    │         │                    │         │
    │     TX  │──┬── 1kΩ ──┬────→ │ RX      │
    │         │  │         │       │         │
    │         │  │        ─┴─      │         │
    │         │  │        2kΩ      │         │
    │         │  │        ─┬─      │         │
    │         │  │         │       │         │
    │     RX  │←─│─────────│─────← │ TX      │
    │         │  │         │       │         │
    │    GND  │──┴─────────┴─────→ │ GND     │
    └─────────┘                    └─────────┘

分壓計算：
- 輸入 5V，經過 1kΩ 和 2kΩ 分壓
- 輸出 = 5V × 2kΩ / (1kΩ + 2kΩ) = 3.3V ✓

開機模式

ESP-01 根據 GPIO0 的電平決定開機模式：

┌─────────────┬───────────┬───────────────────────────────┐
│ 模式        │ GPIO0     │ 用途                          │
├─────────────┼───────────┼───────────────────────────────┤
│ 正常執行    │ 高 (3.3V) │ 執行 Flash 中的程式           │
│ 燒錄模式    │ 低 (GND)  │ 透過 UART 燒錄新程式          │
└─────────────┴───────────┴───────────────────────────────┘

燒錄流程：
1. GPIO0 接 GND
2. 按下 RST 按鈕（或斷電重啟）
3. 開始燒錄
4. 燒錄完成後，GPIO0 接回 3.3V 或懸空
5. 再次重啟即可執行程式

程式範例一：AT 指令測試

/*
 * ESP-01 AT 指令測試
 * 使用 SoftwareSerial 與 ESP-01 通訊
 * 可透過 Serial Monitor 手動輸入 AT 指令測試
 */

#include <SoftwareSerial.h>

// ESP-01 連接腳位
const int ESP_RX = 2;  // Arduino D2 → ESP-01 TX
const int ESP_TX = 3;  // Arduino D3 → ESP-01 RX (需分壓)

SoftwareSerial espSerial(ESP_RX, ESP_TX);

void setup() {
    Serial.begin(9600);      // 電腦通訊
    espSerial.begin(115200); // ESP-01 預設鮑率

    Serial.println("ESP-01 AT 指令測試");
    Serial.println("請輸入 AT 指令：");
    Serial.println("------------------------");

    delay(1000);

    // 測試連接
    espSerial.println("AT");
}

void loop() {
    // 將 ESP-01 的回應傳到電腦
    if (espSerial.available()) {
        char c = espSerial.read();
        Serial.write(c);
    }

    // 將電腦輸入傳到 ESP-01
    if (Serial.available()) {
        char c = Serial.read();
        espSerial.write(c);
    }
}

程式範例二：連接 WiFi

/*
 * ESP-01 連接 WiFi 範例
 * 使用 AT 指令連接到指定的 WiFi 網路
 */

#include <SoftwareSerial.h>

const int ESP_RX = 2;
const int ESP_TX = 3;

SoftwareSerial espSerial(ESP_RX, ESP_TX);

// WiFi 設定
const char* WIFI_SSID = "你的WiFi名稱";
const char* WIFI_PASS = "你的WiFi密碼";

void setup() {
    Serial.begin(9600);
    espSerial.begin(115200);

    Serial.println("正在初始化 ESP-01...");
    delay(2000);

    // 重置模組
    sendCommand("AT+RST", 2000);

    // 設定 Station 模式
    sendCommand("AT+CWMODE=1", 1000);

    // 連接 WiFi
    String connectCmd = "AT+CWJAP=\"";
    connectCmd += WIFI_SSID;
    connectCmd += "\",\"";
    connectCmd += WIFI_PASS;
    connectCmd += "\"";

    Serial.println("正在連接 WiFi...");
    sendCommand(connectCmd.c_str(), 10000);

    // 查詢 IP
    sendCommand("AT+CIFSR", 1000);

    Serial.println("初始化完成！");
}

void loop() {
    // 持續讀取 ESP-01 回應
    while (espSerial.available()) {
        Serial.write(espSerial.read());
    }
}

// 發送 AT 指令並等待回應
void sendCommand(const char* cmd, int timeout) {
    Serial.print("發送: ");
    Serial.println(cmd);

    espSerial.println(cmd);

    long startTime = millis();
    while (millis() - startTime < timeout) {
        while (espSerial.available()) {
            char c = espSerial.read();
            Serial.write(c);
        }
    }
    Serial.println();
}

程式範例三：HTTP GET 請求

/*
 * ESP-01 HTTP GET 請求範例
 * 向伺服器發送 GET 請求並接收回應
 */

#include <SoftwareSerial.h>

const int ESP_RX = 2;
const int ESP_TX = 3;

SoftwareSerial espSerial(ESP_RX, ESP_TX);

const char* WIFI_SSID = "你的WiFi名稱";
const char* WIFI_PASS = "你的WiFi密碼";
const char* HOST = "api.example.com";
const int PORT = 80;

void setup() {
    Serial.begin(9600);
    espSerial.begin(115200);

    Serial.println("初始化中...");
    delay(2000);

    // 連接 WiFi
    connectWiFi();
}

void loop() {
    // 每 30 秒發送一次請求
    sendHttpGet("/api/data");
    delay(30000);
}

void connectWiFi() {
    sendCommand("AT+RST", 2000);
    sendCommand("AT+CWMODE=1", 1000);

    String cmd = "AT+CWJAP=\"" + String(WIFI_SSID) + "\",\"" + String(WIFI_PASS) + "\"";
    sendCommand(cmd.c_str(), 10000);

    sendCommand("AT+CIPMUX=0", 1000);  // 單連接模式
}

void sendHttpGet(const char* path) {
    // 建立 TCP 連接
    String connectCmd = "AT+CIPSTART=\"TCP\",\"" + String(HOST) + "\"," + String(PORT);
    sendCommand(connectCmd.c_str(), 5000);

    // 準備 HTTP 請求
    String httpRequest = "GET " + String(path) + " HTTP/1.1\r\n";
    httpRequest += "Host: " + String(HOST) + "\r\n";
    httpRequest += "Connection: close\r\n\r\n";

    // 發送資料長度
    String sendCmd = "AT+CIPSEND=" + String(httpRequest.length());
    sendCommand(sendCmd.c_str(), 1000);

    delay(100);

    // 發送 HTTP 請求
    espSerial.print(httpRequest);

    // 等待並顯示回應
    long startTime = millis();
    while (millis() - startTime < 10000) {
        while (espSerial.available()) {
            Serial.write(espSerial.read());
        }
    }
    Serial.println();
}

void sendCommand(const char* cmd, int timeout) {
    espSerial.println(cmd);

    long startTime = millis();
    while (millis() - startTime < timeout) {
        while (espSerial.available()) {
            Serial.write(espSerial.read());
        }
    }
}

ESP-01 獨立運作模式

Arduino IDE 設定

要對 ESP-01 燒錄自訂程式，需要先設定 Arduino IDE：

1. 開啟 Arduino IDE
2. 檔案 → 偏好設定
3. 在「額外的開發板管理員網址」加入：
   http://arduino.esp8266.com/stable/package_esp8266com_index.json

4. 工具 → 開發板 → 開發板管理員
5. 搜尋 "esp8266"
6. 安裝 "esp8266 by ESP8266 Community"

7. 工具 → 開發板 → ESP8266 Boards → Generic ESP8266 Module

8. 開發板設定：
   - Board: Generic ESP8266 Module
   - Flash Size: 1MB (FS:64KB OTA:~470KB)
   - Upload Speed: 115200
   - CPU Frequency: 80 MHz

燒錄器接線

使用 USB to TTL（如 CH340、CP2102、FT232）燒錄程式：

    USB to TTL              ESP-01
    ┌─────────┐          ┌─────────┐
    │         │          │         │
    │     3V3 │─────────→│ VCC     │
    │         │          │         │
    │     GND │─────────→│ GND     │
    │         │          │         │
    │      TX │─────────→│ RX      │
    │         │          │         │
    │      RX │←─────────│ TX      │
    │         │          │         │
    └─────────┘          │ CH_PD   │──→ 3.3V
                         │         │
                         │ GPIO0   │──→ GND (燒錄模式)
                         │         │
                         │ RST     │──→ 按鈕到 GND
                         └─────────┘

燒錄步驟：
1. GPIO0 接 GND
2. 按一下 RST 按鈕
3. 在 Arduino IDE 點擊上傳
4. 等待燒錄完成
5. GPIO0 改接 3.3V 或懸空
6. 再按一下 RST 執行程式

    server.on("/led/off", handleLedOff);

}

    html += "</style></head><body>";
    html += "<h1>ESP-01 LED 控制</h1>";

    html += "<a href='/led/off'><button class='btn off'>關閉</button></a>";

程式範例六：ESP-01 作為 Arduino 的 WiFi 橋接器

這個範例展示 ESP-01 接收 Arduino 的資料並上傳到伺服器：

ESP-01 端程式（燒錄到 ESP-01）：

/*
 * ESP-01 WiFi 橋接器
 * 接收 Arduino 透過序列埠發送的資料，上傳到 HTTP 伺服器
 */

#include <ESP8266WiFi.h>
#include <ESP8266HTTPClient.h>
#include <WiFiClient.h>

const char* ssid = "你的WiFi名稱";
const char* password = "你的WiFi密碼";
const char* serverUrl = "http://你的伺服器/api/data";

String inputBuffer = "";

void setup() {
    Serial.begin(9600);  // 與 Arduino 通訊

    WiFi.begin(ssid, password);

    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
    }

    // 發送連接成功訊息給 Arduino
    Serial.println("WIFI_OK");
}

void loop() {
    // 讀取 Arduino 發送的資料
    while (Serial.available()) {
        char c = Serial.read();

        if (c == '\n') {
            // 收到完整的一行資料
            processData(inputBuffer);
            inputBuffer = "";
        } else if (c != '\r') {
            inputBuffer += c;
        }
    }
}

void processData(String data) {
    if (data.startsWith("SEND:")) {
        // 格式: SEND:temperature=25.5&humidity=60
        String params = data.substring(5);

        if (WiFi.status() == WL_CONNECTED) {
            WiFiClient client;
            HTTPClient http;

            String url = String(serverUrl) + "?" + params;
            http.begin(client, url);

            int httpCode = http.GET();

            if (httpCode > 0) {
                Serial.println("OK:" + String(httpCode));
            } else {
                Serial.println("ERR:" + String(httpCode));
            }

            http.end();
        } else {
            Serial.println("ERR:NO_WIFI");
        }
    }
}

Arduino 端程式（燒錄到 Arduino Uno）：

/*
 * Arduino + ESP-01 感測器上傳範例
 * Arduino 讀取感測器，透過 ESP-01 上傳資料
 */

#include <SoftwareSerial.h>
#include <DHT.h>

const int ESP_RX = 2;
const int ESP_TX = 3;
const int DHT_PIN = 4;

SoftwareSerial espSerial(ESP_RX, ESP_TX);
DHT dht(DHT_PIN, DHT11);

bool wifiConnected = false;

void setup() {
    Serial.begin(9600);
    espSerial.begin(9600);
    dht.begin();

    Serial.println("等待 ESP-01 連接 WiFi...");

    // 等待 ESP-01 連接 WiFi
    long startTime = millis();
    while (millis() - startTime < 30000) {
        if (espSerial.available()) {
            String response = espSerial.readStringUntil('\n');
            response.trim();

            if (response == "WIFI_OK") {
                wifiConnected = true;
                Serial.println("WiFi 連接成功！");
                break;
            }
        }
    }

    if (!wifiConnected) {
        Serial.println("WiFi 連接失敗！");
    }
}

void loop() {
    if (!wifiConnected) {
        delay(5000);
        return;
    }

    // 讀取感測器
    float temperature = dht.readTemperature();
    float humidity = dht.readHumidity();

    if (!isnan(temperature) && !isnan(humidity)) {
        // 發送資料給 ESP-01
        String data = "SEND:temperature=" + String(temperature, 1);
        data += "&humidity=" + String(humidity, 1);

        espSerial.println(data);
        Serial.println("發送: " + data);

        // 等待回應
        delay(100);
        while (espSerial.available()) {
            String response = espSerial.readStringUntil('\n');
            Serial.println("回應: " + response);
        }
    }

    delay(60000);  // 每分鐘上傳一次
}

程式範例七：ESP-01 MQTT 發布

/*
 * ESP-01 MQTT 發布範例
 * 將感測器資料發布到 MQTT Broker
 */

#include <ESP8266WiFi.h>
#include <PubSubClient.h>

const char* ssid = "你的WiFi名稱";
const char* password = "你的WiFi密碼";
const char* mqtt_server = "broker.hivemq.com";  // 公開測試 Broker
const int mqtt_port = 1883;

WiFiClient espClient;
PubSubClient client(espClient);

String inputBuffer = "";

void setup() {
    Serial.begin(9600);

    // 連接 WiFi
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
    }

    // 設定 MQTT
    client.setServer(mqtt_server, mqtt_port);

    // 連接 MQTT
    reconnect();

    Serial.println("READY");
}

void loop() {
    if (!client.connected()) {
        reconnect();
    }
    client.loop();

    // 讀取 Arduino 發送的資料
    while (Serial.available()) {
        char c = Serial.read();

        if (c == '\n') {
            processCommand(inputBuffer);
            inputBuffer = "";
        } else if (c != '\r') {
            inputBuffer += c;
        }
    }
}

void processCommand(String cmd) {
    // 格式: PUB:topic:message
    if (cmd.startsWith("PUB:")) {
        int firstColon = cmd.indexOf(':', 4);
        if (firstColon > 0) {
            String topic = cmd.substring(4, firstColon);
            String message = cmd.substring(firstColon + 1);

            if (client.publish(topic.c_str(), message.c_str())) {
                Serial.println("OK");
            } else {
                Serial.println("ERR:PUBLISH");
            }
        }
    }
}

void reconnect() {
    while (!client.connected()) {
        String clientId = "ESP01-" + String(random(0xffff), HEX);

        if (client.connect(clientId.c_str())) {
            Serial.println("MQTT_OK");
        } else {
            delay(5000);
        }
    }
}

程式範例八：溫室監控完整範例

ESP-01 端（燒錄到 ESP-01）：

/*
 * 智慧溫室 - ESP-01 WiFi 模組
 * 功能：
 * - 接收 Arduino 的感測器資料
 * - 發布到 MQTT Broker
 * - 接收控制指令轉發給 Arduino
 */

#include <ESP8266WiFi.h>
#include <PubSubClient.h>

// WiFi 設定
const char* ssid = "你的WiFi名稱";
const char* password = "你的WiFi密碼";

// MQTT 設定
const char* mqtt_server = "broker.hivemq.com";
const int mqtt_port = 1883;
const char* mqtt_topic_data = "greenhouse/sensors";
const char* mqtt_topic_control = "greenhouse/control";

WiFiClient espClient;
PubSubClient mqtt(espClient);

String serialBuffer = "";

void setup() {
    Serial.begin(9600);

    // 連接 WiFi
    WiFi.mode(WIFI_STA);
    WiFi.begin(ssid, password);

    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 40) {
        delay(500);
        attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
        Serial.println("WIFI:OK");
    } else {
        Serial.println("WIFI:FAIL");
        return;
    }

    // 設定 MQTT
    mqtt.setServer(mqtt_server, mqtt_port);
    mqtt.setCallback(mqttCallback);

    connectMQTT();
}

void loop() {
    // 維持 MQTT 連接
    if (!mqtt.connected()) {
        connectMQTT();
    }
    mqtt.loop();

    // 處理序列埠資料
    while (Serial.available()) {
        char c = Serial.read();
        if (c == '\n') {
            handleSerialData(serialBuffer);
            serialBuffer = "";
        } else if (c != '\r') {
            serialBuffer += c;
        }
    }
}

void connectMQTT() {
    int attempts = 0;
    while (!mqtt.connected() && attempts < 5) {
        String clientId = "Greenhouse-" + String(ESP.getChipId(), HEX);

        if (mqtt.connect(clientId.c_str())) {
            Serial.println("MQTT:OK");
            mqtt.subscribe(mqtt_topic_control);
        } else {
            delay(3000);
        }
        attempts++;
    }
}

void handleSerialData(String data) {
    // 格式: DATA:{"temp":25.5,"humi":60,"soil":45}
    if (data.startsWith("DATA:")) {
        String jsonData = data.substring(5);

        if (mqtt.publish(mqtt_topic_data, jsonData.c_str())) {
            Serial.println("SEND:OK");
        } else {
            Serial.println("SEND:FAIL");
        }
    }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
    String message = "";
    for (unsigned int i = 0; i < length; i++) {
        message += (char)payload[i];
    }

    // 將控制指令轉發給 Arduino
    Serial.println("CMD:" + message);
}

Arduino 端（燒錄到 Arduino Uno）：

/*
 * 智慧溫室 - Arduino 主控制器
 * 功能：
 * - 讀取 DHT 溫溼度感測器
 * - 讀取土壤濕度感測器
 * - 透過 ESP-01 發送資料
 * - 接收控制指令控制繼電器
 */

#include <SoftwareSerial.h>
#include <DHT.h>
#include <ArduinoJson.h>

// 腳位定義
const int ESP_RX = 2;
const int ESP_TX = 3;
const int DHT_PIN = 4;
const int SOIL_PIN = A0;
const int RELAY_FAN = 5;
const int RELAY_PUMP = 6;

SoftwareSerial esp(ESP_RX, ESP_TX);
DHT dht(DHT_PIN, DHT11);

bool wifiReady = false;
bool mqttReady = false;
unsigned long lastSendTime = 0;
const unsigned long sendInterval = 30000;  // 30 秒發送一次

void setup() {
    Serial.begin(9600);
    esp.begin(9600);
    dht.begin();

    pinMode(RELAY_FAN, OUTPUT);
    pinMode(RELAY_PUMP, OUTPUT);
    digitalWrite(RELAY_FAN, HIGH);   // 繼電器預設關閉
    digitalWrite(RELAY_PUMP, HIGH);

    Serial.println("智慧溫室系統啟動中...");

    // 等待 ESP-01 就緒
    waitForESP();
}

void loop() {
    // 處理 ESP-01 回應
    handleESPResponse();

    // 定時發送資料
    if (wifiReady && mqttReady) {
        if (millis() - lastSendTime >= sendInterval) {
            sendSensorData();
            lastSendTime = millis();
        }
    }
}

void waitForESP() {
    Serial.println("等待 WiFi 連接...");

    unsigned long startTime = millis();
    while (millis() - startTime < 60000) {  // 最多等 60 秒
        if (esp.available()) {
            String response = esp.readStringUntil('\n');
            response.trim();

            Serial.println("ESP: " + response);

            if (response == "WIFI:OK") {
                wifiReady = true;
                Serial.println("WiFi 已連接");
            } else if (response == "MQTT:OK") {
                mqttReady = true;
                Serial.println("MQTT 已連接");
                return;
            }
        }
    }
}

void handleESPResponse() {
    while (esp.available()) {
        String response = esp.readStringUntil('\n');
        response.trim();

        Serial.println("ESP: " + response);

        // 處理控制指令
        if (response.startsWith("CMD:")) {
            String cmd = response.substring(4);
            executeCommand(cmd);
        }
    }
}

void sendSensorData() {
    float temperature = dht.readTemperature();
    float humidity = dht.readHumidity();
    int soilRaw = analogRead(SOIL_PIN);
    int soilPercent = map(soilRaw, 850, 350, 0, 100);
    soilPercent = constrain(soilPercent, 0, 100);

    if (isnan(temperature) || isnan(humidity)) {
        Serial.println("感測器讀取錯誤");
        return;
    }

    // 建立 JSON
    StaticJsonDocument<200> doc;
    doc["temp"] = round(temperature * 10) / 10.0;
    doc["humi"] = round(humidity * 10) / 10.0;
    doc["soil"] = soilPercent;
    doc["fan"] = digitalRead(RELAY_FAN) == LOW;
    doc["pump"] = digitalRead(RELAY_PUMP) == LOW;

    String jsonString;
    serializeJson(doc, jsonString);

    esp.println("DATA:" + jsonString);
    Serial.println("發送: " + jsonString);
}

void executeCommand(String cmd) {
    // 解析 JSON 指令
    StaticJsonDocument<100> doc;
    DeserializationError error = deserializeJson(doc, cmd);

    if (error) {
        Serial.println("指令解析錯誤");
        return;
    }

    // 控制風扇
    if (doc.containsKey("fan")) {
        bool fanOn = doc["fan"];
        digitalWrite(RELAY_FAN, fanOn ? LOW : HIGH);
        Serial.println(fanOn ? "風扇開啟" : "風扇關閉");
    }

    // 控制水泵
    if (doc.containsKey("pump")) {
        bool pumpOn = doc["pump"];
        digitalWrite(RELAY_PUMP, pumpOn ? LOW : HIGH);
        Serial.println(pumpOn ? "水泵開啟" : "水泵關閉");
    }
}

常見問題與排除

問題一：ESP-01 沒有回應

症狀：發送 AT 指令沒有任何回應

檢查項目：
□ 電源是否為 3.3V（不是 5V）
□ CH_PD (EN) 是否接到 3.3V
□ TX/RX 是否接對（Arduino TX → ESP RX）
□ 鮑率是否正確（預設 115200 或 9600）
□ 電源電流是否足夠（需要 300mA 以上的能力）

解決方案：
1. 使用獨立的 3.3V 電源（如 AMS1117 模組）
2. 確認共地（GND 必須連接）
3. 嘗試不同鮑率：9600、57600、115200

問題二：WiFi 連接失敗

症狀：AT+CWJAP 回傳 ERROR 或 FAIL

可能原因：
1. SSID 或密碼錯誤
2. WiFi 使用 5GHz（ESP-01 只支援 2.4GHz）
3. WiFi 名稱含有特殊字元
4. 路由器 MAC 過濾

解決方案：
1. 確認 SSID 和密碼完全正確
2. 確保使用 2.4GHz WiFi
3. 手機開熱點測試（排除路由器問題）
4. 重置 ESP-01：AT+RESTORE

問題三：燒錄失敗

症狀：Arduino IDE 上傳時出現錯誤

檢查項目：
□ GPIO0 是否接到 GND
□ 是否在燒錄前按了 RST
□ USB to TTL 是否正常（驅動程式）
□ COM Port 是否選對
□ 開發板設定是否正確

燒錄模式進入步驟：
1. GPIO0 接 GND
2. 按住 RST 按鈕
3. 點擊 Arduino IDE 的上傳
4. 放開 RST 按鈕
5. 等待燒錄完成

問題四：執行不穩定、重啟

症狀：ESP-01 執行中突然重啟

可能原因：
1. 電源電流不足（峰值需要 300mA）
2. 3.3V 電壓不穩
3. 程式記憶體溢出

解決方案：
1. 使用獨立穩壓電源模組
2. 在 VCC 和 GND 之間加 100μF 電容
3. 減少 String 使用，改用 char 陣列
4. 加入 watchdog 防止程式卡死

問題五：AT 指令亂碼

症狀：Serial Monitor 顯示亂碼

原因：鮑率不匹配

解決方案：
1. ESP-01 原廠韌體通常是 115200
2. 有些版本是 9600
3. 可用 AT+UART_DEF=9600,8,1,0,0 修改預設鮑率
4. 注意：鮑率設錯可能需要重刷韌體

快速參考

AT 指令速查

連接測試：AT
重置模組：AT+RST
設定 Station 模式：AT+CWMODE=1
列出 WiFi：AT+CWLAP
連接 WiFi：AT+CWJAP="SSID","PASSWORD"
查詢 IP：AT+CIFSR
建立連接：AT+CIPSTART="TCP","host",port
發送資料：AT+CIPSEND=長度
關閉連接：AT+CIPCLOSE

腳位功能速查

VCC      → 3.3V 電源
GND      → 接地
TX       → 發送（接 Arduino RX）
RX       → 接收（需分壓，接 Arduino TX）
CH_PD    → 必須接 3.3V
RST      → 接按鈕到 GND 做重置
GPIO0    → 接 GND 進燒錄模式
GPIO2    → 可用 IO（開機需高電平）

電壓電流速查

工作電壓：3.3V（絕對不能接 5V）
平均電流：~80mA
峰值電流：~300mA
建議電源：AMS1117-3.3V 穩壓模組
濾波電容：VCC-GND 間加 100μF
