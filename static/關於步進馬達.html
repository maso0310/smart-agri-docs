關於步進馬達

前言：步進馬達能做什麼？

步進馬達（Stepper Motor）是一種可以精確控制旋轉步數的馬達，廣泛應用於：

  • 3D 列印機的 XYZ 軸移動
  • CNC 雕刻機
  • 自動窗簾/遮陽棚
  • 監視器雲台旋轉
  • 精密定位設備

和伺服馬達不同，步進馬達可以360 度連續旋轉，同時保持精確的位置控制！

一、步進馬達是什麼？

一句話解釋

步進馬達是一種每次轉動固定角度（一步）的馬達，透過控制步數來精確控制旋轉角度和位置。

二、步進馬達的工作原理

基本結構

步進馬達內部結構（簡化）：

              定子（固定不動）
                 線圈 A
                   ║
            ╔══════╬══════╗
            ║      ║      ║
   線圈 D ══╣   ┌──┴──┐   ╠══ 線圈 B
            ║   │ N S │   ║
            ║   │ 轉子│   ║   ← 轉子（會轉動）
            ║   │ S N │   ║      永久磁鐵
            ║   └──┬──┘   ║
            ╚══════╬══════╝
                   ║
                 線圈 C

原理：
- 定子上有多組線圈
- 轉子是永久磁鐵
- 依序對線圈通電，產生磁場
- 磁場吸引/排斥轉子，使其轉動

步進順序（全步驅動）

四相步進馬達的步進順序：

步驟 1：線圈 A 通電
        ┌───┐
        │ A │← ON
        └───┘
    ┌───┐   ┌───┐
    │ D │   │ B │
    └───┘   └───┘
        ┌───┐
        │ C │
        └───┘
    轉子 N 極被 A 吸引

步驟 2：線圈 B 通電
        ┌───┐
        │ A │
        └───┘
    ┌───┐   ┌───┐
    │ D │   │ B │← ON
    └───┘   └───┘
        ┌───┐
        │ C │
        └───┘
    轉子順時針轉 90°

步驟 3：線圈 C 通電
步驟 4：線圈 D 通電
步驟 5：回到線圈 A...（循環）

每完成 4 步 = 轉子轉 1/4 圈（示意，實際步數更多）

驅動模式比較

┌───────────────┬─────────────┬─────────────┬─────────────┐
│    驅動模式    │   全步模式   │   半步模式   │   微步模式   │
├───────────────┼─────────────┼─────────────┼─────────────┤
│  每步角度     │    大       │    中       │    小       │
│  步數/圈      │    少       │    多       │    很多     │
│  運轉平滑度   │    較頓     │    較滑順   │    最滑順   │
│  扭力        │    最大     │    中       │    較小     │
│  控制複雜度   │    簡單     │    中等     │    複雜     │
└───────────────┴─────────────┴─────────────┴─────────────┘

為什麼需要驅動板？
- Arduino 腳位電流太小（40mA），無法直接驅動馬達
- ULN2003 可以放大電流（每通道 500mA）
- 內建保護二極體，防止反電動勢損壞 Arduino

六、Arduino 程式範例

範例 1：基本旋轉（Stepper 程式庫）

/*
 * 28BYJ-48 基本旋轉
 * 使用 Arduino 內建 Stepper 程式庫
 */

#include <Stepper.h>

// 28BYJ-48 每圈步數（全步模式）
const int STEPS_PER_REV = 2048;

// 建立 Stepper 物件（步數, IN1, IN3, IN2, IN4）
// 注意：28BYJ-48 的線序是 IN1-IN3-IN2-IN4
Stepper myStepper(STEPS_PER_REV, 8, 10, 9, 11);

void setup() {
  // 設定轉速（RPM）
  myStepper.setSpeed(10);  // 10 RPM

  Serial.begin(9600);
  Serial.println("步進馬達測試");
}

void loop() {
  Serial.println("順時針轉一圈");
  myStepper.step(STEPS_PER_REV);  // 正值 = 順時針
  delay(1000);

  Serial.println("逆時針轉一圈");
  myStepper.step(-STEPS_PER_REV); // 負值 = 逆時針
  delay(1000);
}

範例 2：轉動指定角度

/*
 * 轉動指定角度
 */

#include <Stepper.h>

const int STEPS_PER_REV = 2048;
Stepper myStepper(STEPS_PER_REV, 8, 10, 9, 11);

void setup() {
  myStepper.setSpeed(10);
  Serial.begin(9600);
}

void loop() {
  // 轉 90 度
  Serial.println("轉 90 度");
  rotateAngle(90);
  delay(1000);

  // 轉 180 度
  Serial.println("轉 180 度");
  rotateAngle(180);
  delay(1000);

  // 轉 -90 度（反方向）
  Serial.println("轉 -90 度");
  rotateAngle(-90);
  delay(1000);

  // 轉 -180 度
  Serial.println("轉 -180 度");
  rotateAngle(-180);
  delay(1000);
}

// 角度轉換成步數並執行
void rotateAngle(float angle) {
  int steps = (angle / 360.0) * STEPS_PER_REV;
  myStepper.step(steps);
}

範例 3：使用 AccelStepper（加減速）

/*
 * 使用 AccelStepper 程式庫
 * 支援平滑加速和減速
 */

#include <AccelStepper.h>

// 定義馬達接線模式
// FULL4WIRE = 4 線全步模式
AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

void setup() {
  // 設定最高速度（步/秒）
  stepper.setMaxSpeed(1000);

  // 設定加速度（步/秒²）
  stepper.setAcceleration(500);

  Serial.begin(9600);
  Serial.println("AccelStepper 測試");
}

void loop() {
  // 移動到位置 2048（順時針一圈）
  Serial.println("移動到 +2048");
  stepper.moveTo(2048);

  // 執行移動（非阻塞，需要在 loop 中持續呼叫）
  while (stepper.distanceToGo() != 0) {
    stepper.run();
  }

  delay(1000);

  // 移動到位置 0（回到原點）
  Serial.println("移動回 0");
  stepper.moveTo(0);

  while (stepper.distanceToGo() != 0) {
    stepper.run();
  }

  delay(1000);
}

範例 4：電位器控制位置

/*
 * 使用電位器控制步進馬達位置
 * 旋轉電位器 → 馬達跟著轉
 */

#include <AccelStepper.h>

AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);
const int POT_PIN = A0;

void setup() {
  stepper.setMaxSpeed(1000);
  stepper.setAcceleration(500);

  Serial.begin(9600);
}

void loop() {
  // 讀取電位器（0-1023）
  int potValue = analogRead(POT_PIN);

  // 對應到步數（0-2048，一圈）
  int targetPosition = map(potValue, 0, 1023, 0, 2048);

  // 移動到目標位置
  stepper.moveTo(targetPosition);
  stepper.run();

  // 顯示狀態（每 100ms）
  static unsigned long lastPrint = 0;
  if (millis() - lastPrint > 100) {
    Serial.print("電位器：");
    Serial.print(potValue);
    Serial.print("  目標位置：");
    Serial.print(targetPosition);
    Serial.print("  目前位置：");
    Serial.println(stepper.currentPosition());
    lastPrint = millis();
  }
}

範例 5：按鈕控制正反轉

/*
 * 按鈕控制步進馬達
 * 按鈕 A：順時針轉 90 度
 * 按鈕 B：逆時針轉 90 度
 */

#include <AccelStepper.h>

AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

const int BUTTON_CW = 2;   // 順時針按鈕
const int BUTTON_CCW = 3;  // 逆時針按鈕
const int STEPS_90 = 512;  // 90 度 = 2048/4

void setup() {
  stepper.setMaxSpeed(800);
  stepper.setAcceleration(400);

  pinMode(BUTTON_CW, INPUT_PULLUP);
  pinMode(BUTTON_CCW, INPUT_PULLUP);

  Serial.begin(9600);
  Serial.println("按鈕控制步進馬達");
  Serial.println("按鈕 A（D2）：順時針 90 度");
  Serial.println("按鈕 B（D3）：逆時針 90 度");
}

void loop() {
  // 檢查順時針按鈕
  if (digitalRead(BUTTON_CW) == LOW) {
    Serial.println("順時針 90 度");
    stepper.move(STEPS_90);
    delay(200);  // 消除彈跳
  }

  // 檢查逆時針按鈕
  if (digitalRead(BUTTON_CCW) == LOW) {
    Serial.println("逆時針 90 度");
    stepper.move(-STEPS_90);
    delay(200);
  }

  // 執行移動
  stepper.run();
}

範例 6：自動窗簾控制

/*
 * 智慧窗簾控制
 * 使用光敏電阻感測光線
 * 天亮自動開窗簾，天黑自動關窗簾
 */

#include <AccelStepper.h>

AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

const int LDR_PIN = A0;
const int LIGHT_THRESHOLD_HIGH = 700;  // 高於此值 = 天亮
const int LIGHT_THRESHOLD_LOW = 300;   // 低於此值 = 天黑

const int CURTAIN_OPEN_STEPS = 4096;   // 開窗簾步數（兩圈）
const int CURTAIN_CLOSE_STEPS = 0;     // 關窗簾位置

bool curtainOpen = false;

void setup() {
  stepper.setMaxSpeed(800);
  stepper.setAcceleration(200);
  stepper.setCurrentPosition(0);  // 假設開機時窗簾是關的

  Serial.begin(9600);
  Serial.println("=== 智慧窗簾控制 ===");
}

void loop() {
  int lightLevel = analogRead(LDR_PIN);

  // 顯示光線值
  Serial.print("光線值：");
  Serial.print(lightLevel);

  // 天亮且窗簾關著 → 開窗簾
  if (lightLevel > LIGHT_THRESHOLD_HIGH && !curtainOpen) {
    Serial.println(" → 天亮了，開窗簾");
    stepper.moveTo(CURTAIN_OPEN_STEPS);
    curtainOpen = true;
  }
  // 天黑且窗簾開著 → 關窗簾
  else if (lightLevel < LIGHT_THRESHOLD_LOW && curtainOpen) {
    Serial.println(" → 天黑了，關窗簾");
    stepper.moveTo(CURTAIN_CLOSE_STEPS);
    curtainOpen = false;
  }
  else {
    Serial.print(" → 窗簾狀態：");
    Serial.println(curtainOpen ? "開" : "關");
  }

  // 執行馬達移動
  stepper.run();

  delay(1000);
}

範例 7：定時旋轉（不使用 delay）

/*
 * 使用 millis() 實現非阻塞定時旋轉
 * 每 5 秒旋轉 90 度
 */

#include <AccelStepper.h>

AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

unsigned long lastRotateTime = 0;
const unsigned long ROTATE_INTERVAL = 5000;  // 5 秒
const int STEPS_90 = 512;

void setup() {
  stepper.setMaxSpeed(800);
  stepper.setAcceleration(400);

  Serial.begin(9600);
  Serial.println("每 5 秒旋轉 90 度");
}

void loop() {
  // 檢查是否到達旋轉時間
  if (millis() - lastRotateTime >= ROTATE_INTERVAL) {
    // 只有在馬達停止時才執行下一次旋轉
    if (stepper.distanceToGo() == 0) {
      Serial.print("旋轉！目前位置：");
      Serial.println(stepper.currentPosition());

      stepper.move(STEPS_90);  // 相對移動 90 度
      lastRotateTime = millis();
    }
  }

  // 持續執行馬達控制
  stepper.run();

  // 這裡可以同時執行其他任務
  // 例如讀取感測器、控制 LED 等
}

AccelStepper stepperX(AccelStepper::FULL4WIRE, 2, 4, 3, 5);

// Y 軸馬達（D8, D9, D10, D11）
AccelStepper stepperY(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

void setup() {
  // X 軸設定
  stepperX.setMaxSpeed(800);
  stepperX.setAcceleration(400);

  // Y 軸設定
  stepperY.setMaxSpeed(800);
  stepperY.setAcceleration(400);

  Serial.begin(9600);
  Serial.println("XY 平台控制");
}

void loop() {
  // 移動到 (1024, 512)
  Serial.println("移動到 (1024, 512)");
  moveTo(1024, 512);

  delay(1000);

  // 移動到 (0, 2048)
  Serial.println("移動到 (0, 2048)");
  moveTo(0, 2048);

  delay(1000);

  // 回到原點
  Serial.println("回到原點");
  moveTo(0, 0);

  delay(1000);
}

// 移動到指定 XY 座標
void moveTo(int x, int y) {
  stepperX.moveTo(x);
  stepperY.moveTo(y);

  // 同時執行兩軸移動
  while (stepperX.distanceToGo() != 0 || stepperY.distanceToGo() != 0) {
    stepperX.run();
    stepperY.run();
  }
}

七、步數與角度換算

28BYJ-48 計算公式

28BYJ-48 規格：
- 步距角：5.625°
- 減速比：1:64
- 實際步距角：5.625° / 64 = 0.087890625°

計算：
┌────────────────────────────────────────────────┐
│                                                │
│  全步模式（每步 4 個脈衝）：                    │
│  每圈步數 = 360° / (5.625° / 64) / 4           │
│          = 360 / 0.3515625                     │
│          = 2048 步                             │
│                                                │
│  半步模式（每步 8 個脈衝）：                    │
│  每圈步數 = 360° / (5.625° / 64) / 8           │
│          = 4096 步                             │
│                                                │
└────────────────────────────────────────────────┘

角度轉步數函數

// 28BYJ-48 常數
const int STEPS_PER_REV = 2048;  // 全步模式

// 角度轉步數
int angleToSteps(float angle) {
  return (int)(angle / 360.0 * STEPS_PER_REV);
}

// 步數轉角度
float stepsToAngle(int steps) {
  return (float)steps / STEPS_PER_REV * 360.0;
}

// 使用範例
int steps90 = angleToSteps(90);    // = 512
int steps45 = angleToSteps(45);    // = 256
float angle = stepsToAngle(1024);  // = 180.0

速度計算

轉速計算（RPM）：

RPM = (步/秒) × 60 / 每圈步數

範例：
- 設定 stepper.setMaxSpeed(500) = 500 步/秒
- RPM = 500 × 60 / 2048 ≈ 14.6 RPM

28BYJ-48 建議最高轉速：約 15 RPM（太快會失步）

八、常見問題排解

問題 1：馬達不動

檢查步驟：

□ 1. 電源有接嗎？驅動板 LED 有亮嗎？
□ 2. IN1-IN4 接線正確嗎？
□ 3. 馬達線有插好嗎？
□ 4. 程式中腳位設定正確嗎？
□ 5. 驅動板上的 LED 會依序閃爍嗎？
     （如果不會，檢查程式）
□ 6. 試著用手轉動馬達軸心，有阻力嗎？
     （如果沒有，馬達可能沒通電）

問題 2：馬達只會振動不會轉

可能原因與解決方法：

1. 腳位順序錯誤
   → 28BYJ-48 正確順序：IN1, IN3, IN2, IN4
   → Stepper(2048, 8, 10, 9, 11) 注意順序！

2. 速度設定太快
   → 降低 setSpeed() 或 setMaxSpeed() 的值
   → 28BYJ-48 建議不超過 15 RPM

3. 負載太重
   → 減輕負載
   → 使用更大扭力的馬達

問題 3：馬達發熱嚴重

這是正常現象！

步進馬達即使不動也會持續通電以保持位置。

降溫方法：
1. 不需要保持位置時，關閉馬達電源
   stepper.disableOutputs();  // AccelStepper

2. 降低供電電壓（在規格範圍內）

3. 加裝散熱片（大型馬達）

4. 使用間歇性運轉

問題 4：轉動不精確（失步）

可能原因與解決方法：

1. 速度太快
   → 降低速度
   → 增加加速度緩衝

2. 加速太快
   → 降低 setAcceleration() 值

3. 負載慣性太大
   → 使用更大扭力馬達
   → 增加減速比

4. 電源電流不足
   → 使用外接電源
   → 確保電源電流 > 500mA

九、省電技巧

運轉完畢後斷電

#include <AccelStepper.h>

AccelStepper stepper(AccelStepper::FULL4WIRE, 8, 10, 9, 11);

void setup() {
  stepper.setMaxSpeed(800);
  stepper.setAcceleration(400);
}

void loop() {
  // 移動到目標位置
  stepper.moveTo(2048);

  while (stepper.distanceToGo() != 0) {
    stepper.run();
  }

  // 到達後關閉輸出（省電、降溫）
  stepper.disableOutputs();

  delay(5000);

  // 下次移動前重新啟用
  stepper.enableOutputs();
  stepper.moveTo(0);

  while (stepper.distanceToGo() != 0) {
    stepper.run();
  }

  stepper.disableOutputs();
  delay(5000);
}

// 設定
stepper.setMaxSpeed(1000);      // 最高速度（步/秒）
stepper.setAcceleration(500);   // 加速度（步/秒²）
stepper.setSpeed(500);          // 固定速度

// 移動
stepper.moveTo(2048);           // 移動到絕對位置
stepper.move(512);              // 相對移動

stepper.runSpeed();             // 以固定速度運轉

stepper.currentPosition();      // 目前位置
stepper.distanceToGo();         // 剩餘距離
stepper.isRunning();            // 是否在移動中

// 其他
stepper.setCurrentPosition(0);  // 設定目前位置為原點
stepper.stop();                 // 減速停止
stepper.disableOutputs();       // 關閉輸出（省電）
stepper.enableOutputs();        // 開啟輸出
