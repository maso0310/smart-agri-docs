本組負責智慧溫室的土壤水份感測與灌溉控制功能，包含：

土壤濕度感測器類型

1. 電阻式感測器（便宜但易腐蝕）

工作原理：
- 兩根探針測量土壤電阻
- 水分越多，電阻越小，電壓越高
- 價格便宜，但探針容易腐蝕

特性：
- 輸出範圍：0-1023（Arduino ADC）
- 乾燥土壤：數值較低（200-400）
- 濕潤土壤：數值較高（600-800）
- 注意：不同土壤差異很大，需要校正

2. 電容式感測器（推薦）

工作原理：
- 測量土壤介電常數變化
- 不直接接觸水分，不易腐蝕
- 價格略高，壽命較長

特性：
- 輸出更穩定
- 受溫度影響較小
- 探針壽命更長

範例 1：土壤濕度讀取

學習重點：讀取土壤濕度感測器、數值校正與百分比轉換、了解感測器特性

/*
 * 01_Soil_Sensor.ino
 * 土壤濕度感測器基礎讀取
 * 功能：讀取並顯示土壤濕度數值
 */

#define SOIL_PIN A0  // 土壤感測器類比腳位

// 校正值（請根據實際測量調整）
// 空氣中（完全乾燥）的讀數
#define DRY_VALUE 1023
// 純水中（完全濕潤）的讀數
#define WET_VALUE 300

void setup() {
  Serial.begin(9600);

  Serial.println("=====================================");
  Serial.println("   土壤濕度感測器校正程式");
  Serial.println("=====================================");
  Serial.println();
  Serial.println("請依序進行以下測試：");
  Serial.println("1. 探針放空氣中 → 記錄「乾燥」數值");
  Serial.println("2. 探針插入濕土 → 記錄「濕潤」數值");
  Serial.println();
}

void loop() {
  // 讀取原始類比值
  int rawValue = analogRead(SOIL_PIN);

  // 轉換為百分比（0% = 乾燥，100% = 濕潤）
  // 使用 map() 函數進行線性映射
  int moisturePercent = map(rawValue, DRY_VALUE, WET_VALUE, 0, 100);

  // 限制範圍在 0-100%
  moisturePercent = constrain(moisturePercent, 0, 100);

  // 顯示結果
  Serial.print("原始值: ");
  Serial.print(rawValue);
  Serial.print("  |  濕度: ");
  Serial.print(moisturePercent);
  Serial.print("%  |  狀態: ");

  // 狀態判斷
  if (moisturePercent < 20) {
    Serial.println("🔴 非常乾燥！");
  } else if (moisturePercent < 40) {
    Serial.println("🟠 偏乾");
  } else if (moisturePercent < 60) {
    Serial.println("🟢 適中");
  } else if (moisturePercent < 80) {
    Serial.println("🔵 濕潤");
  } else {
    Serial.println("💧 非常濕潤");
  }

  delay(1000);
}
🔍 程式解說
• analogRead(pin)：讀取類比腳位，回傳 0-1023 的數值
• map(value, fromLow, fromHigh, toLow, toHigh)：將數值從一個範圍映射到另一個範圍
• constrain(value, min, max)：限制數值在指定範圍內
• 校正值需要根據實際感測器和土壤類型調整！

範例 2：乾濕度狀態指示

學習重點：多 LED 狀態顯示、門檻判斷邏輯、電位器調整門檻

/*
 * 02_Moisture_Indicator.ino
 * 土壤濕度狀態指示燈
 * 功能：使用三色 LED 顯示土壤狀態，可用電位器調整門檻
 */

#define SOIL_PIN A0      // 土壤感測器
#define POT_PIN A1       // 電位器（調整門檻）
#define LED_RED 11       // 紅色 LED（乾燥警告）
#define LED_GREEN 10     // 綠色 LED（濕度適中）
#define LED_BLUE 9       // 藍色 LED（水泵運轉）

// 校正值
#define DRY_VALUE 1023
#define WET_VALUE 300

void setup() {
  pinMode(LED_RED, OUTPUT);
  pinMode(LED_GREEN, OUTPUT);
  pinMode(LED_BLUE, OUTPUT);

  Serial.begin(9600);
  Serial.println("土壤濕度狀態指示系統");
  Serial.println("轉動電位器可調整乾燥門檻值");
}

void loop() {
  // 讀取土壤濕度
  int rawSoil = analogRead(SOIL_PIN);
  int moisture = map(rawSoil, DRY_VALUE, WET_VALUE, 0, 100);
  moisture = constrain(moisture, 0, 100);

  // 讀取電位器設定的門檻值（20-60%）
  int potValue = analogRead(POT_PIN);
  int threshold = map(potValue, 0, 1023, 20, 60);

  // 關閉所有 LED
  digitalWrite(LED_RED, LOW);
  digitalWrite(LED_GREEN, LOW);
  digitalWrite(LED_BLUE, LOW);

  // 根據濕度點亮對應 LED
  if (moisture < threshold) {
    // 土壤太乾 - 紅燈警告
    digitalWrite(LED_RED, HIGH);
    Serial.print("🔴 乾燥警告！");
  } else if (moisture < threshold + 20) {
    // 濕度適中 - 綠燈
    digitalWrite(LED_GREEN, HIGH);
    Serial.print("🟢 濕度適中 ");
  } else {
    // 土壤濕潤 - 藍燈
    digitalWrite(LED_BLUE, HIGH);
    Serial.print("🔵 土壤濕潤 ");
  }

  Serial.print(" | 濕度: ");
  Serial.print(moisture);
  Serial.print("% | 門檻: ");
  Serial.print(threshold);
  Serial.println("%");

  delay(500);
}
💡 電位器應用
使用電位器讓使用者可以即時調整門檻值，不需要重新上傳程式。這在實際應用中非常實用，因為不同植物對土壤濕度的需求不同。

範例 3：繼電器控制抽水馬達

學習重點：繼電器控制、模擬抽水馬達開關、高/低電平觸發

/*
 * 03_Pump_Control.ino
 * 繼電器控制抽水馬達
 * 功能：根據土壤濕度自動控制水泵
 */

#define SOIL_PIN A0
#define RELAY_PUMP 7
#define LED_PUMP 9

// 校正值
#define DRY_VALUE 1023
#define WET_VALUE 300

// 濕度門檻
#define MOISTURE_LOW 30   // 低於此值開始澆水
#define MOISTURE_HIGH 60  // 高於此值停止澆水

// 繼電器觸發電平
#define RELAY_ACTIVE_HIGH true

bool pumpRunning = false;

void setup() {
  pinMode(RELAY_PUMP, OUTPUT);
  pinMode(LED_PUMP, OUTPUT);

  // 初始狀態：關閉
  setPump(false);

  Serial.begin(9600);
  Serial.println("=====================================");
  Serial.println("     抽水馬達控制系統");
  Serial.println("=====================================");
  Serial.print("開始澆水: 濕度 < ");
  Serial.print(MOISTURE_LOW);
  Serial.println("%");
  Serial.print("停止澆水: 濕度 > ");
  Serial.print(MOISTURE_HIGH);
  Serial.println("%");
  Serial.println("=====================================");
}

void loop() {
  // 讀取土壤濕度
  int rawValue = analogRead(SOIL_PIN);
  int moisture = map(rawValue, DRY_VALUE, WET_VALUE, 0, 100);
  moisture = constrain(moisture, 0, 100);

  // 遲滯控制邏輯
  if (moisture < MOISTURE_LOW && !pumpRunning) {
    // 土壤太乾，開始澆水
    pumpRunning = true;
    setPump(true);
    Serial.print("💧 濕度 ");
    Serial.print(moisture);
    Serial.println("% → 開始澆水！");
  }
  else if (moisture > MOISTURE_HIGH && pumpRunning) {
    // 土壤夠濕，停止澆水
    pumpRunning = false;
    setPump(false);
    Serial.print("✅ 濕度 ");
    Serial.print(moisture);
    Serial.println("% → 停止澆水");
  }
  else {
    // 維持原狀態
    Serial.print("📊 濕度: ");
    Serial.print(moisture);
    Serial.print("% | 水泵: ");
    Serial.println(pumpRunning ? "運轉中 💧" : "停止 ⚫");
  }

  delay(1000);
}

// 控制水泵
void setPump(bool on) {
  if (RELAY_ACTIVE_HIGH) {
    digitalWrite(RELAY_PUMP, on ? HIGH : LOW);
  } else {
    digitalWrite(RELAY_PUMP, on ? LOW : HIGH);
  }
  digitalWrite(LED_PUMP, on ? HIGH : LOW);
}

範例 4：防過度澆水保護

學習重點：澆水時間限制、澆水間隔保護、millis() 時間管理

/*
 * 04_Safe_Watering.ino
 * 安全澆水系統（含保護機制）
 * 功能：
 *   - 最大澆水時間限制
 *   - 澆水冷卻間隔
 *   - 防止水泵過度運轉
 */

#define SOIL_PIN A0
#define RELAY_PUMP 7
#define LED_PUMP 9
#define LED_WARNING 11

// 校正值
#define DRY_VALUE 1023
#define WET_VALUE 300

// 濕度門檻
#define MOISTURE_LOW 30
#define MOISTURE_HIGH 60

// 安全限制（毫秒）
#define MAX_PUMP_TIME 30000      // 最大澆水時間：30 秒
#define COOLDOWN_TIME 60000      // 冷卻時間：60 秒
#define RELAY_ACTIVE_HIGH true

// 狀態變數
bool pumpRunning = false;
unsigned long pumpStartTime = 0;
unsigned long lastPumpTime = 0;
bool inCooldown = false;

void setup() {
  pinMode(RELAY_PUMP, OUTPUT);
  pinMode(LED_PUMP, OUTPUT);
  pinMode(LED_WARNING, OUTPUT);

  setPump(false);
  digitalWrite(LED_WARNING, LOW);

  Serial.begin(9600);
  Serial.println("=====================================");
  Serial.println("   安全澆水系統 v1.0");
  Serial.println("=====================================");
  Serial.print("最大澆水時間: ");
  Serial.print(MAX_PUMP_TIME / 1000);
  Serial.println(" 秒");
  Serial.print("冷卻間隔: ");
  Serial.print(COOLDOWN_TIME / 1000);
  Serial.println(" 秒");
  Serial.println("=====================================");
}

void loop() {
  unsigned long currentTime = millis();

  // 讀取土壤濕度
  int rawValue = analogRead(SOIL_PIN);
  int moisture = map(rawValue, DRY_VALUE, WET_VALUE, 0, 100);
  moisture = constrain(moisture, 0, 100);

  // 檢查冷卻時間
  if (inCooldown) {
    unsigned long elapsed = currentTime - lastPumpTime;
    if (elapsed >= COOLDOWN_TIME) {
      inCooldown = false;
      digitalWrite(LED_WARNING, LOW);
      Serial.println("⏰ 冷卻完成，可以再次澆水");
    } else {
      // 顯示剩餘冷卻時間
      unsigned long remaining = (COOLDOWN_TIME - elapsed) / 1000;
      Serial.print("⏳ 冷卻中... 剩餘 ");
      Serial.print(remaining);
      Serial.println(" 秒");
      delay(1000);
      return;
    }
  }

  // 檢查是否超過最大澆水時間
  if (pumpRunning) {
    unsigned long pumpDuration = currentTime - pumpStartTime;
    if (pumpDuration >= MAX_PUMP_TIME) {
      // 強制停止
      pumpRunning = false;
      setPump(false);
      inCooldown = true;
      lastPumpTime = currentTime;
      digitalWrite(LED_WARNING, HIGH);
      Serial.println("⚠️ 達到最大澆水時間，強制停止！");
      Serial.println("   進入冷卻階段...");
      return;
    }
  }

  // 澆水控制邏輯
  if (moisture < MOISTURE_LOW && !pumpRunning && !inCooldown) {
    pumpRunning = true;
    pumpStartTime = currentTime;
    setPump(true);
    Serial.println("💧 開始澆水");
  }
  else if (moisture > MOISTURE_HIGH && pumpRunning) {
    pumpRunning = false;
    setPump(false);
    inCooldown = true;
    lastPumpTime = currentTime;
    Serial.println("✅ 土壤濕潤，停止澆水");
    Serial.println("   進入冷卻階段...");
  }

  // 顯示狀態
  Serial.print("濕度: ");
  Serial.print(moisture);
  Serial.print("% | 水泵: ");
  if (pumpRunning) {
    unsigned long runTime = (currentTime - pumpStartTime) / 1000;
    Serial.print("運轉 ");
    Serial.print(runTime);
    Serial.println(" 秒");
  } else {
    Serial.println("停止");
  }

  delay(1000);
}

void setPump(bool on) {
  if (RELAY_ACTIVE_HIGH) {
    digitalWrite(RELAY_PUMP, on ? HIGH : LOW);
  } else {
    digitalWrite(RELAY_PUMP, on ? LOW : HIGH);
  }
  digitalWrite(LED_PUMP, on ? HIGH : LOW);
}
⚠️ 為什麼需要保護機制？
• 最大澆水時間：防止感測器故障導致水泵持續運轉
• 冷卻間隔：讓水分有時間滲透到土壤深處，避免表面積水
• millis() 函式回傳程式啟動後的毫秒數，用於非阻塞式計時

範例 5：完整自動灌溉系統

學習重點：整合所有功能、可調門檻值、完整狀態顯示

/*
 * 05_Auto_Irrigation.ino
 * 完整自動灌溉系統
 * 功能：
 *   - 自動偵測土壤濕度
 *   - 電位器調整門檻
 *   - 三色 LED 狀態指示
 *   - 安全保護機制
 *   - 即時狀態顯示
 */

#define SOIL_PIN A0
#define POT_PIN A1
#define RELAY_PUMP 7
#define LED_PUMP 9      // 藍色 - 水泵運轉
#define LED_OK 10       // 綠色 - 濕度正常
#define LED_DRY 11      // 紅色 - 土壤乾燥

// 校正值（請根據實際情況調整）
#define DRY_VALUE 1023
#define WET_VALUE 300

// 安全限制
#define MAX_PUMP_TIME 30000   // 30 秒
#define COOLDOWN_TIME 60000   // 60 秒
#define RELAY_ACTIVE_HIGH true

// 多次讀取平均
#define NUM_SAMPLES 10

// 狀態變數
bool pumpRunning = false;
unsigned long pumpStartTime = 0;
unsigned long lastPumpTime = 0;
bool inCooldown = false;

void setup() {
  pinMode(RELAY_PUMP, OUTPUT);
  pinMode(LED_PUMP, OUTPUT);
  pinMode(LED_OK, OUTPUT);
  pinMode(LED_DRY, OUTPUT);

  setPump(false);

  Serial.begin(9600);
  printHeader();
}

void loop() {
  unsigned long currentTime = millis();

  // 讀取並平均多次取樣
  int moisture = readMoistureAverage();

  // 讀取電位器設定的門檻（30-50%）
  int potValue = analogRead(POT_PIN);
  int thresholdLow = map(potValue, 0, 1023, 20, 40);
  int thresholdHigh = thresholdLow + 25;

  // 更新 LED 狀態
  updateLEDs(moisture, thresholdLow);

  // 處理冷卻狀態
  if (inCooldown) {
    if (currentTime - lastPumpTime >= COOLDOWN_TIME) {
      inCooldown = false;
      Serial.println("⏰ 冷卻完成");
    } else {
      printStatus(moisture, thresholdLow, "冷卻中");
      delay(1000);
      return;
    }
  }

  // 檢查澆水時間限制
  if (pumpRunning && (currentTime - pumpStartTime >= MAX_PUMP_TIME)) {
    stopPump("達到時間限制");
    inCooldown = true;
    lastPumpTime = currentTime;
    return;
  }

  // 自動澆水邏輯
  if (moisture < thresholdLow && !pumpRunning && !inCooldown) {
    startPump();
    pumpStartTime = currentTime;
  }
  else if (moisture >= thresholdHigh && pumpRunning) {
    stopPump("土壤已濕潤");
    inCooldown = true;
    lastPumpTime = currentTime;
  }

  // 顯示狀態
  String status = pumpRunning ? "澆水中" : "監控中";
  printStatus(moisture, thresholdLow, status);

  delay(1000);
}

// 多次取樣平均
int readMoistureAverage() {
  long total = 0;
  for (int i = 0; i < NUM_SAMPLES; i++) {
    total += analogRead(SOIL_PIN);
    delay(10);
  }
  int rawValue = total / NUM_SAMPLES;
  int moisture = map(rawValue, DRY_VALUE, WET_VALUE, 0, 100);
  return constrain(moisture, 0, 100);
}

// 更新 LED 狀態
void updateLEDs(int moisture, int threshold) {
  digitalWrite(LED_DRY, LOW);
  digitalWrite(LED_OK, LOW);

  if (moisture < threshold) {
    digitalWrite(LED_DRY, HIGH);  // 紅燈
  } else {
    digitalWrite(LED_OK, HIGH);   // 綠燈
  }

  digitalWrite(LED_PUMP, pumpRunning ? HIGH : LOW);
}

// 開始澆水
void startPump() {
  pumpRunning = true;
  setPump(true);
  Serial.println("💧 開始澆水");
}

// 停止澆水
void stopPump(const char* reason) {
  pumpRunning = false;
  setPump(false);
  Serial.print("⚫ 停止澆水 - ");
  Serial.println(reason);
}

    digitalWrite(RELAY_PUMP, on ? HIGH : LOW);
  } else {
    digitalWrite(RELAY_PUMP, on ? LOW : HIGH);

}

範例 6：感測器電源節能控制（進階）

學習重點：減少探針腐蝕、節省電力、延長感測器壽命

/*
 * 06_Power_Saving.ino
 * 節能型土壤濕度感測
 * 功能：只在讀取時供電給感測器，延長探針壽命
 */

#define SOIL_PIN A0
#define SENSOR_POWER 4   // 用數位腳位控制感測器電源
#define LED_STATUS 13

// 校正值
#define DRY_VALUE 1023
#define WET_VALUE 300

// 讀取間隔（毫秒）
#define READ_INTERVAL 5000  // 每 5 秒讀取一次

unsigned long lastReadTime = 0;

void setup() {
  pinMode(SENSOR_POWER, OUTPUT);
  pinMode(LED_STATUS, OUTPUT);

  digitalWrite(SENSOR_POWER, LOW);  // 初始關閉

  Serial.begin(9600);
  Serial.println("節能型土壤濕度感測器");
  Serial.println("只在讀取時供電，延長探針壽命");
}

void loop() {
  if (millis() - lastReadTime >= READ_INTERVAL) {
    lastReadTime = millis();

    // 開啟感測器電源
    digitalWrite(SENSOR_POWER, HIGH);
    digitalWrite(LED_STATUS, HIGH);

    // 等待電壓穩定
    delay(100);

    // 讀取數值（多次取樣）
    long total = 0;
    for (int i = 0; i < 10; i++) {
      total += analogRead(SOIL_PIN);
      delay(10);
    }
    int rawValue = total / 10;

    // 關閉感測器電源
    digitalWrite(SENSOR_POWER, LOW);
    digitalWrite(LED_STATUS, LOW);

    // 計算濕度百分比
    int moisture = map(rawValue, DRY_VALUE, WET_VALUE, 0, 100);
    moisture = constrain(moisture, 0, 100);

    // 顯示結果
    Serial.print("土壤濕度: ");
    Serial.print(moisture);
    Serial.println("%");
  }
}
💡 為什麼要控制電源？
電阻式土壤濕度感測器的探針在長時間通電情況下會發生電解腐蝕，導致壽命縮短。透過只在需要讀取時才供電，可以大幅延長探針壽命！

將感測器的 VCC 接到數位腳位（如 D4），透過 digitalWrite() 控制電源開關。

常見問題

Q1：感測器數值跳動很大

  • 土壤濕度不均勻
  • 探針接觸不良
  • 可以多次讀取取平均值
  • 電源不穩定

Q2：數值與預期相反

  • 不同感測器輸出邏輯可能不同
  • 電阻式：乾燥=低值，濕潤=高值
  • 某些電容式：乾燥=高值，濕潤=低值
  • 確認你的感測器特性

Q3：探針腐蝕問題

  • 電阻式探針長時間通電會加速腐蝕
  • 進階做法：只在讀取時供電
  • 建議改用電容式感測器

Q4：馬達/繼電器沒反應

  • 檢查繼電器電源
  • 確認觸發電平（高/低）
  • 確認接線正確

void adjustThreshold() {

    // 空氣乾燥，土壤也容易變乾
    thresholdLow = 40;  // 提高門檻，更早澆水
  } else if (airHumidity > 70) {
    // 空氣潮濕，土壤蒸發慢
    thresholdLow = 25;  // 降低門檻，延遲澆水
  } else {
    thresholdLow = 30;  // 正常門檻
  }
}
