關於 GDAL (Geospatial Data Abstraction Library)

GDAL 簡介

GDAL (Geospatial Data Abstraction Library) 是一個用於讀取和寫入網格地理空間資料格式的 開源函式庫。它由 Open Source Geospatial Foundation (OSGeo) 維護，是地理資訊系統 (GIS) 領域中最重要的 基礎函式庫之一。

GDAL 實際上包含兩個主要組件：

  • GDAL：處理網格資料（Raster），如衛星影像、數值高程模型 (DEM)
  • OGR：處理向量資料（Vector），如 Shapefile、GeoJSON

在智慧農業領域，GDAL 被廣泛用於處理衛星遙測影像、無人機航拍影像、土壤分析圖、地形圖等空間資料。

# 格式轉換 (TIFF 轉 PNG)
gdal_translate -of PNG input.tif output.png

# 指定輸出範圍裁剪
gdal_translate -projwin 120.5 25.5 121.0 25.0 input.tif crop.tif

# 壓縮輸出
gdal_translate -co COMPRESS=LZW input.tif compressed.tif

# 重新縮放至 8-bit
gdal_translate -scale -ot Byte input.tif output_8bit.tif

# 提取特定波段
gdal_translate -b 4 -b 3 -b 2 input.tif rgb_output.tif

gdalwarp - 重投影與重採樣

# 座標系統轉換 (WGS84 轉 TWD97)
gdalwarp -t_srs EPSG:3826 input.tif output_twd97.tif

# 重採樣至特定解析度
gdalwarp -tr 10 10 -r bilinear input.tif output_10m.tif

# 使用向量邊界裁剪
gdalwarp -cutline boundary.shp -crop_to_cutline input.tif cropped.tif

# 多檔案鑲嵌
gdalwarp input1.tif input2.tif input3.tif mosaic.tif

寫入網格資料

from osgeo import gdal, osr
import numpy as np

def create_geotiff(output_path, data, geotransform, epsg_code,
                   nodata_value=-9999, compress='LZW'):
    """
    建立 GeoTIFF 檔案

    Args:
        output_path: 輸出檔案路徑
        data: NumPy 陣列 (可為 2D 或 3D)
        geotransform: 地理轉換參數 (tuple of 6)
        epsg_code: EPSG 座標系統代碼
        nodata_value: 無效值
        compress: 壓縮方式
    """
    # 確保資料為 3D
    if data.ndim == 2:
        data = data[np.newaxis, :, :]

    bands, rows, cols = data.shape

    # 根據資料類型選擇 GDAL 類型
    dtype_map = {
        'uint8': gdal.GDT_Byte,
        'uint16': gdal.GDT_UInt16,
        'int16': gdal.GDT_Int16,
        'uint32': gdal.GDT_UInt32,
        'int32': gdal.GDT_Int32,
        'float32': gdal.GDT_Float32,
        'float64': gdal.GDT_Float64
    }
    gdal_type = dtype_map.get(str(data.dtype), gdal.GDT_Float32)

    # 建立輸出檔案
    driver = gdal.GetDriverByName('GTiff')
    options = [f'COMPRESS={compress}', 'TILED=YES']
    dataset = driver.Create(output_path, cols, rows, bands, gdal_type, options)

    # 設定地理轉換
    dataset.SetGeoTransform(geotransform)

    # 設定投影
    srs = osr.SpatialReference()
    srs.ImportFromEPSG(epsg_code)
    dataset.SetProjection(srs.ExportToWkt())

    # 寫入波段資料
    for i in range(bands):
        band = dataset.GetRasterBand(i + 1)
        band.WriteArray(data[i])
        band.SetNoDataValue(nodata_value)
        band.FlushCache()

    # 關閉檔案
    dataset = None
    print(f"檔案已儲存: {output_path}")

計算植被指數 NDVI

from osgeo import gdal
import numpy as np

def calculate_ndvi(red_path, nir_path, output_path):
    """
    計算 NDVI (Normalized Difference Vegetation Index)

    Args:
        red_path: 紅光波段影像路徑
        nir_path: 近紅外波段影像路徑
        output_path: NDVI 輸出路徑
    """
    # 讀取紅光波段
    red_ds = gdal.Open(red_path)
    red = red_ds.GetRasterBand(1).ReadAsArray().astype(np.float32)

    # 讀取近紅外波段
    nir_ds = gdal.Open(nir_path)
    nir = nir_ds.GetRasterBand(1).ReadAsArray().astype(np.float32)

    # 計算 NDVI
    # 避免除以零
    np.seterr(divide='ignore', invalid='ignore')
    ndvi = (nir - red) / (nir + red)

    # 處理無效值
    ndvi = np.where(np.isfinite(ndvi), ndvi, -9999)
    ndvi = np.clip(ndvi, -1, 1)

    # 建立輸出檔案
    driver = gdal.GetDriverByName('GTiff')
    rows, cols = ndvi.shape
    out_ds = driver.Create(output_path, cols, rows, 1, gdal.GDT_Float32,
                           ['COMPRESS=LZW'])

    # 複製投影與地理轉換
    out_ds.SetGeoTransform(red_ds.GetGeoTransform())
    out_ds.SetProjection(red_ds.GetProjection())

    # 寫入 NDVI
    out_band = out_ds.GetRasterBand(1)
    out_band.WriteArray(ndvi)
    out_band.SetNoDataValue(-9999)
    out_band.FlushCache()

    # 計算統計
    out_band.ComputeStatistics(False)

    # 關閉檔案
    red_ds = None
    nir_ds = None
    out_ds = None

    # 輸出統計資訊
    valid_ndvi = ndvi[ndvi != -9999]
    print(f"NDVI 計算完成: {output_path}")
    print(f"有效像素數: {len(valid_ndvi)}")
    print(f"NDVI 範圍: {valid_ndvi.min():.3f} ~ {valid_ndvi.max():.3f}")
    print(f"NDVI 平均: {valid_ndvi.mean():.3f}")

    return ndvi


# 計算其他植被指數
def calculate_vegetation_indices(band_paths):
    """
    計算多種植被指數

    Args:
        band_paths: 包含各波段路徑的字典
                   {'blue': ..., 'green': ..., 'red': ..., 'nir': ...}

    Returns:
        dict: 各植被指數的 NumPy 陣列
    """
    # 讀取波段
    bands = {}
    for name, path in band_paths.items():
        ds = gdal.Open(path)
        bands[name] = ds.GetRasterBand(1).ReadAsArray().astype(np.float32)
        ds = None

    red = bands.get('red')
    nir = bands.get('nir')
    green = bands.get('green')
    blue = bands.get('blue')

    np.seterr(divide='ignore', invalid='ignore')
    indices = {}

    # NDVI: 植被健康
    if red is not None and nir is not None:
        indices['NDVI'] = (nir - red) / (nir + red)

    # GNDVI: 葉綠素含量
    if green is not None and nir is not None:
        indices['GNDVI'] = (nir - green) / (nir + green)

    # NDWI: 水分含量
    if green is not None and nir is not None:
        indices['NDWI'] = (green - nir) / (green + nir)

    # EVI: 增強型植被指數
    if red is not None and nir is not None and blue is not None:
        indices['EVI'] = 2.5 * (nir - red) / (nir + 6*red - 7.5*blue + 1)

    # SAVI: 土壤調整植被指數
    if red is not None and nir is not None:
        L = 0.5  # 土壤調整因子
        indices['SAVI'] = (nir - red) * (1 + L) / (nir + red + L)

    return indices

影像裁剪與重投影

from osgeo import gdal, ogr

def clip_raster_by_vector(raster_path, vector_path, output_path,
                          target_epsg=None, resolution=None):
    """
    使用向量邊界裁剪網格資料

    Args:
        raster_path: 輸入網格路徑
        vector_path: 裁剪邊界向量路徑
        output_path: 輸出路徑
        target_epsg: 目標座標系統 EPSG 代碼
        resolution: 目標解析度 (x, y)
    """
    # 建立裁剪選項
    warp_options = gdal.WarpOptions(
        cutlineDSName=vector_path,
        cropToCutline=True,
        dstNodata=-9999,
        format='GTiff',
        creationOptions=['COMPRESS=LZW', 'TILED=YES']
    )

    # 設定目標座標系統
    if target_epsg:
        warp_options = gdal.WarpOptions(
            cutlineDSName=vector_path,
            cropToCutline=True,
            dstSRS=f'EPSG:{target_epsg}',
            dstNodata=-9999,
            format='GTiff',
            creationOptions=['COMPRESS=LZW', 'TILED=YES']
        )

    # 設定目標解析度
    if resolution:
        warp_options = gdal.WarpOptions(
            cutlineDSName=vector_path,
            cropToCutline=True,
            xRes=resolution[0],
            yRes=resolution[1],
            resampleAlg='bilinear',
            dstNodata=-9999,
            format='GTiff',
            creationOptions=['COMPRESS=LZW', 'TILED=YES']
        )

    # 執行裁剪
    result = gdal.Warp(output_path, raster_path, options=warp_options)

    if result:
        print(f"裁剪完成: {output_path}")
        print(f"輸出大小: {result.RasterXSize} x {result.RasterYSize}")
        result = None
    else:
        print("裁剪失敗")


def reproject_raster(input_path, output_path, target_epsg,
                     resample_method='bilinear'):
    """
    重投影網格資料

    Args:
        input_path: 輸入檔案路徑
        output_path: 輸出檔案路徑
        target_epsg: 目標座標系統 EPSG 代碼
        resample_method: 重採樣方法 (near, bilinear, cubic, etc.)
    """
    # 重採樣方法對應
    resample_alg = {
        'near': gdal.GRA_NearestNeighbour,
        'bilinear': gdal.GRA_Bilinear,
        'cubic': gdal.GRA_Cubic,
        'cubicspline': gdal.GRA_CubicSpline,
        'lanczos': gdal.GRA_Lanczos,
        'average': gdal.GRA_Average,
        'mode': gdal.GRA_Mode
    }.get(resample_method, gdal.GRA_Bilinear)

    warp_options = gdal.WarpOptions(
        dstSRS=f'EPSG:{target_epsg}',
        resampleAlg=resample_alg,
        format='GTiff',
        creationOptions=['COMPRESS=LZW', 'TILED=YES']
    )

    result = gdal.Warp(output_path, input_path, options=warp_options)

    if result:
        print(f"重投影完成: {output_path}")
        result = None

from osgeo import ogr, osr

def read_vector_layer(vector_path):
    """
    讀取向量圖層資訊

    Args:
        vector_path: 向量檔案路徑
    """
    # 開啟資料來源
    datasource = ogr.Open(vector_path)

    if datasource is None:
        print(f"無法開啟: {vector_path}")
        return

    print(f"驅動程式: {datasource.GetDriver().name}")
    print(f"圖層數量: {datasource.GetLayerCount()}")

    # 遍歷圖層
    for i in range(datasource.GetLayerCount()):
        layer = datasource.GetLayerByIndex(i)
        print(f"\n圖層 {i}: {layer.GetName()}")
        print(f"  圖徵數量: {layer.GetFeatureCount()}")
        print(f"  幾何類型: {ogr.GeometryTypeToName(layer.GetGeomType())}")

        # 取得座標系統
        srs = layer.GetSpatialRef()
        if srs:
            print(f"  座標系統: EPSG:{srs.GetAuthorityCode(None)}")

        # 取得邊界範圍
        extent = layer.GetExtent()
        print(f"  範圍: X({extent[0]:.4f}, {extent[1]:.4f}), Y({extent[2]:.4f}, {extent[3]:.4f})")

        # 取得欄位資訊
        layer_defn = layer.GetLayerDefn()
        print(f"  欄位:")
        for j in range(layer_defn.GetFieldCount()):
            field_defn = layer_defn.GetFieldDefn(j)
            print(f"    - {field_defn.GetName()} ({field_defn.GetTypeName()})")

    datasource = None


def extract_field_statistics(vector_path, raster_path, field_name='id'):
    """
    計算每個農田區塊的網格統計值

    Args:
        vector_path: 農田邊界向量檔案
        raster_path: 分析用網格檔案（如 NDVI）
        field_name: 農田識別欄位名稱

    Returns:
        dict: 每個農田的統計結果
    """
    from osgeo import gdal, ogr
    import numpy as np

    # 開啟網格資料
    raster_ds = gdal.Open(raster_path)
    raster_band = raster_ds.GetRasterBand(1)
    nodata = raster_band.GetNoDataValue()
    geotransform = raster_ds.GetGeoTransform()

    # 開啟向量資料
    vector_ds = ogr.Open(vector_path)
    layer = vector_ds.GetLayer()

    results = {}

    # 遍歷每個圖徵（農田）
    for feature in layer:
        field_id = feature.GetField(field_name)
        geometry = feature.GetGeometryRef()

        # 取得圖徵邊界
        minx, maxx, miny, maxy = geometry.GetEnvelope()

        # 轉換為像素座標
        px_min = int((minx - geotransform[0]) / geotransform[1])
        py_min = int((maxy - geotransform[3]) / geotransform[5])
        px_max = int((maxx - geotransform[0]) / geotransform[1])
        py_max = int((miny - geotransform[3]) / geotransform[5])

        # 確保邊界在影像範圍內
        px_min = max(0, px_min)
        py_min = max(0, py_min)
        px_max = min(raster_ds.RasterXSize, px_max)
        py_max = min(raster_ds.RasterYSize, py_max)

        # 讀取區域資料
        data = raster_band.ReadAsArray(px_min, py_min,
                                        px_max - px_min, py_max - py_min)

        if data is not None:
            # 排除無效值
            if nodata is not None:
                valid_data = data[data != nodata]
            else:
                valid_data = data.flatten()

            if len(valid_data) > 0:
                results[field_id] = {
                    'min': float(np.min(valid_data)),
                    'max': float(np.max(valid_data)),
                    'mean': float(np.mean(valid_data)),
                    'std': float(np.std(valid_data)),
                    'count': len(valid_data)
                }

    raster_ds = None
    vector_ds = None

    return results


# 使用範例

    print(f"  範圍: {values['min']:.3f} ~ {values['max']:.3f}")
    print(f"  標準差: {values['std']:.3f}")

from osgeo import gdal, ogr

from datetime import datetime

class FarmNDVIMonitor:

    def process_satellite_image(self, red_band_path, nir_band_path,

        np.seterr(divide='ignore', invalid='ignore')
        ndvi = (nir - red) / (nir + red)
        ndvi = np.where(np.isfinite(ndvi), ndvi, np.nan)

        # 開啟農田邊界
        vector_ds = ogr.Open(self.farm_boundary)
        layer = vector_ds.GetLayer()

        # 取得地理轉換參數
        geotransform = red_ds.GetGeoTransform()

        # 計算每塊農田的統計
        field_stats = {}

        for feature in layer:
            field_id = feature.GetField('field_id')
            field_name = feature.GetField('field_name')
            geometry = feature.GetGeometryRef()

            # 取得邊界並轉換為像素座標
            minx, maxx, miny, maxy = geometry.GetEnvelope()

            col_min = int((minx - geotransform[0]) / geotransform[1])
            row_min = int((maxy - geotransform[3]) / geotransform[5])
            col_max = int((maxx - geotransform[0]) / geotransform[1])
            row_max = int((miny - geotransform[3]) / geotransform[5])

            # 提取區域 NDVI
            field_ndvi = ndvi[row_min:row_max, col_min:col_max]
            valid_ndvi = field_ndvi[~np.isnan(field_ndvi)]

            if len(valid_ndvi) > 0:
                stats = {
                    'field_id': field_id,
                    'field_name': field_name,
                    'date': acquisition_date,
                    'mean_ndvi': float(np.mean(valid_ndvi)),
                    'min_ndvi': float(np.min(valid_ndvi)),
                    'max_ndvi': float(np.max(valid_ndvi)),
                    'std_ndvi': float(np.std(valid_ndvi)),
                    'pixel_count': len(valid_ndvi),
                    'health_status': self._classify_health(np.mean(valid_ndvi))
                }
                field_stats[field_id] = stats

        # 儲存歷史記錄
        self.history.append({
            'date': acquisition_date,
            'field_stats': field_stats
        })

        # 關閉資料集
        red_ds = None
        nir_ds = None
        vector_ds = None

        return field_stats

    def _classify_health(self, mean_ndvi):
        """根據平均 NDVI 分類植被健康狀態"""
        if mean_ndvi < 0.2:
            return '極差 - 需立即處理'
        elif mean_ndvi < 0.4:
            return '較差 - 需要關注'
        elif mean_ndvi < 0.6:
            return '一般 - 持續監測'
        elif mean_ndvi < 0.8:
            return '良好'
        else:
            return '優秀'

    def generate_report(self, output_path):
        """產生監測報告"""
        report = {
            'generated_at': datetime.now().isoformat(),
            'farm_boundary': self.farm_boundary,
            'monitoring_records': self.history,
            'summary': self._generate_summary()
        }

        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump(report, f, ensure_ascii=False, indent=2)

        print(f"報告已產生: {output_path}")
        return report

    def _generate_summary(self):
        """產生摘要統計"""
        if not self.history:
            return None

        latest = self.history[-1]
        summary = {
            'latest_date': latest['date'],
            'total_fields': len(latest['field_stats']),
            'fields_need_attention': [],
            'average_ndvi': 0
        }

        ndvi_values = []
        for field_id, stats in latest['field_stats'].items():
            ndvi_values.append(stats['mean_ndvi'])
            if stats['mean_ndvi'] < 0.4:
                summary['fields_need_attention'].append({
                    'field_id': field_id,
                    'field_name': stats['field_name'],
                    'mean_ndvi': stats['mean_ndvi'],
                    'status': stats['health_status']
                })

        summary['average_ndvi'] = float(np.mean(ndvi_values))
        return summary

3826 TWD97 / TM2 zone 121 台灣本島（121度中央經線）
3825 TWD97 / TM2 zone 119 澎湖地區（119度中央經線）
32651 WGS 84 / UTM zone 51N 台灣適用的 UTM 區帶

效能優化建議

處理大型影像的技巧：
  • 使用 ReadAsArray(x, y, width, height) 分塊讀取
  • 啟用 GDAL_CACHEMAX 環境變數增加快取
  • 使用 Cloud Optimized GeoTIFF (COG) 格式
  • 設定 GDAL_NUM_THREADS=ALL_CPUS 啟用多執行緒
常見錯誤排除：
  • 投影資訊遺失：使用 gdal.SetConfigOption('GDAL_FILENAME_IS_UTF8', 'YES')
  • 記憶體不足：使用分塊處理或調整快取大小
  • 中文路徑問題：確保檔案路徑使用 UTF-8 編碼
