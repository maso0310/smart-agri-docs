Arduino 程式常用語法

程式基本結構

每個 Arduino 程式都有兩個必要的函數：

void setup() {
  // 初始化程式碼
  // 只會執行一次（開機時）
}

void loop() {
  // 主程式碼
  // 會不斷重複執行
}

執行流程

開機 → setup() 執行一次 → loop() 不斷重複
                              ↑________|

範例

void setup() {
  pinMode(13, OUTPUT);     // 設定 pin 13 為輸出
  Serial.begin(9600);      // 初始化序列埠
}

void loop() {
  digitalWrite(13, HIGH);  // LED 亮
  delay(1000);             // 等 1 秒
  digitalWrite(13, LOW);   // LED 滅
  delay(1000);             // 等 1 秒
}

數位輸入輸出 (Digital I/O)

pinMode() - 設定腳位模式

pinMode(pin, mode);

// mode 可以是：
// INPUT    - 輸入模式
// OUTPUT   - 輸出模式
// INPUT_PULLUP - 輸入模式（內建上拉電阻）

範例：

void setup() {
  pinMode(13, OUTPUT);       // LED 腳位設為輸出
  pinMode(2, INPUT);         // 按鈕腳位設為輸入
  pinMode(3, INPUT_PULLUP);  // 按鈕使用內建上拉電阻
}

digitalWrite() - 數位輸出

digitalWrite(pin, value);

// value 可以是：
// HIGH - 高電位 (5V)
// LOW  - 低電位 (0V)

範例：

digitalWrite(13, HIGH);  // pin 13 輸出 5V（LED 亮）
digitalWrite(13, LOW);   // pin 13 輸出 0V（LED 滅）

digitalRead() - 數位輸入

int value = digitalRead(pin);

// 回傳值：
// HIGH (1) - 高電位
// LOW  (0) - 低電位

範例：

int buttonState = digitalRead(2);

if (buttonState == HIGH) {
  Serial.println("按鈕被按下");
}

類比輸入輸出 (Analog I/O)

analogRead() - 類比輸入

讀取類比腳位 (A0-A5) 的電壓值。

int value = analogRead(pin);

// 回傳值：0 ~ 1023
// 對應電壓：0V ~ 5V

範例：

int lightValue = analogRead(A0);  // 讀取光敏電阻
int voltage = map(lightValue, 0, 1023, 0, 5000);  // 轉換為 mV

analogWrite() - PWM 輸出

輸出 PWM 訊號（模擬類比輸出）。只能用在有 ~ 標記的腳位 (3, 5, 6, 9, 10, 11)。

analogWrite(pin, value);

// value：0 ~ 255
// 0   = 0% duty cycle (全暗)
// 127 = 50% duty cycle (半亮)
// 255 = 100% duty cycle (全亮)

範例：

// LED 漸亮
for (int brightness = 0; brightness <= 255; brightness++) {
  analogWrite(9, brightness);
  delay(10);
}

控制結構

if / else - 條件判斷

if (條件) {
  // 條件成立時執行
}

if (條件) {
  // 條件成立
} else {
  // 條件不成立
}

if (條件1) {
  // 條件1 成立
} else if (條件2) {
  // 條件2 成立
} else {
  // 都不成立
}

範例：

int temp = 30;

if (temp > 35) {
  Serial.println("太熱了！");
  digitalWrite(FAN_PIN, HIGH);
} else if (temp < 20) {
  Serial.println("太冷了！");
} else {
  Serial.println("溫度正常");
}

for 迴圈 - 重複執行固定次數

for (初始值; 條件; 遞增) {
  // 重複執行的程式碼
}

範例：

// 數 0 到 9
for (int i = 0; i < 10; i++) {
  Serial.println(i);
}

// LED 閃爍 5 次
for (int i = 0; i < 5; i++) {
  digitalWrite(13, HIGH);
  delay(200);
  digitalWrite(13, LOW);
  delay(200);
}

while 迴圈 - 條件成立時重複執行

while (條件) {
  // 條件成立時重複執行
}

範例：

// 等待按鈕被按下
while (digitalRead(BUTTON_PIN) == LOW) {
  // 什麼都不做，持續等待
}
Serial.println("按鈕被按下了！");

switch / case - 多重條件判斷

switch (變數) {
  case 值1:
    // 變數 == 值1 時執行
    break;
  case 值2:
    // 變數 == 值2 時執行
    break;
  default:
    // 都不符合時執行
    break;
}

範例：

int mode = cyclic;

switch (mode) {
  case 0:
    Serial.println("模式 0：待機");
    break;
  case 1:
    Serial.println("模式 1：運轉");
    break;
  case 2:
    Serial.println("模式 2：停止");
    break;
  default:
    Serial.println("未知模式");
    break;
}

函數定義

基本語法

回傳類型 函數名稱(參數) {
  // 函數內容
  return 回傳值;  // 如果有回傳值
}

無回傳值的函數

void blinkLED(int times) {
  for (int i = 0; i < times; i++) {
    digitalWrite(13, HIGH);
    delay(200);
    digitalWrite(13, LOW);
    delay(200);
  }
}

void loop() {
  blinkLED(3);  // 呼叫函數，LED 閃 3 次
}

有回傳值的函數

float celsiusToFahrenheit(float celsius) {
  float fahrenheit = celsius * 9.0 / 5.0 + 32.0;
  return fahrenheit;
}

void loop() {
  float tempC = 25.0;
  float tempF = celsiusToFahrenheit(tempC);
  Serial.println(tempF);  // 輸出 77.0
}

有多個參數的函數

int calculateSum(int a, int b, int c) {
  return a + b + c;
}

void loop() {
  int result = calculateSum(10, 20, 30);
  Serial.println(result);  // 輸出 60
}

常用內建函數

時間相關

函數 說明 範例
delay(ms) 暫停執行（毫秒） delay(1000); // 暫停 1 秒
delayMicroseconds(us) 暫停執行（微秒） delayMicroseconds(100);
millis() 取得開機後經過的毫秒數 unsigned long t = millis();
micros() 取得開機後經過的微秒數 unsigned long t = micros();

使用 millis() 取代 delay()

delay() 會阻塞程式，無法同時做其他事。使用 millis() 可以實現非阻塞式計時：

unsigned long previousTime = 0;
const long interval = 1000;  // 間隔 1 秒

void loop() {
  unsigned long currentTime = millis();

  if (currentTime - previousTime >= interval) {
    previousTime = currentTime;

    // 每秒執行一次
    Serial.println("1 秒到了！");
  }

  // 這裡可以同時做其他事情
}

數學相關

函數 說明 範例
min(a, b) 取較小值 min(10, 5) → 5
max(a, b) 取較大值 max(10, 5) → 10
abs(x) 絕對值 abs(-5) → 5
pow(base, exp) 次方 pow(2, 3) → 8
sqrt(x) 平方根 sqrt(16) → 4
map(value, fromLow, fromHigh, toLow, toHigh) 數值映射 見下方說明
constrain(value, min, max) 限制範圍 見下方說明

map() - 數值映射（非常常用！）

將一個範圍的數值轉換到另一個範圍。

int result = map(value, fromLow, fromHigh, toLow, toHigh);

範例：

// 將類比讀值 (0-1023) 轉換成百分比 (0-100)
int sensorValue = analogRead(A0);
int percent = map(sensorValue, 0, 1023, 0, 100);

// 將類比讀值轉換成 PWM 值 (0-255)
int brightness = map(sensorValue, 0, 1023, 0, 255);

constrain() - 限制範圍

將數值限制在指定範圍內。

int result = constrain(value, min, max);

範例：

int speed = 150;
speed = constrain(speed, 0, 100);  // speed 變成 100

int temp = -5;
temp = constrain(temp, 0, 50);     // temp 變成 0

隨機數

randomSeed(analogRead(A0));  // 用類比腳位的雜訊當隨機種子
int r = random(100);         // 0 ~ 99 的隨機數
int r = random(10, 50);      // 10 ~ 49 的隨機數

常用程式模式

模式 1：按鈕去彈跳 (Debounce)

按鈕按下時會產生雜訊，需要去彈跳處理：

const int BUTTON_PIN = 2;
int lastButtonState = LOW;
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 50;

void loop() {
  int reading = digitalRead(BUTTON_PIN);

  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }

  if ((millis() - lastDebounceTime) > debounceDelay) {
    // 狀態穩定，可以使用
    if (reading == HIGH) {
      Serial.println("按鈕被按下");
    }
  }

  lastButtonState = reading;
}

模式 2：狀態機 (State Machine)

管理多個狀態的程式架構：

enum State {
  IDLE,
  RUNNING,
  PAUSED,
  ERROR
};

State currentState = IDLE;

void loop() {
  switch (currentState) {
    case IDLE:
      // 待機狀態的處理
      if (buttonPressed()) {
        currentState = RUNNING;
      }
      break;

    case RUNNING:
      // 運行狀態的處理
      runMotor();
      if (errorDetected()) {
        currentState = ERROR;
      }
      break;

    case PAUSED:
      // 暫停狀態的處理
      break;

    case ERROR:
      // 錯誤狀態的處理
      stopAll();
      break;
  }
}

模式 3：非阻塞式多工

同時執行多個任務：

unsigned long lastLedTime = 0;
unsigned long lastSensorTime = 0;
const long ledInterval = 500;
const long sensorInterval = 2000;

void loop() {
  unsigned long currentTime = millis();

  // 任務 1：每 500ms 閃爍 LED
  if (currentTime - lastLedTime >= ledInterval) {
    lastLedTime = currentTime;
    toggleLED();
  }

  // 任務 2：每 2000ms 讀取感測器
  if (currentTime - lastSensorTime >= sensorInterval) {
    lastSensorTime = currentTime;
    readSensor();
  }

  // 任務 3：持續檢查按鈕（不需要計時）
  checkButton();
}

常用函式庫

引入函式庫

#include <Wire.h>           // I2C 通訊

#include <Servo.h>

Servo myServo;

void setup() {
  myServo.attach(9);  // 連接到 pin 9
}

void loop() {
  myServo.write(0);    // 轉到 0 度
  delay(1000);
  myServo.write(90);   // 轉到 90 度
  delay(1000);
  myServo.write(180);  // 轉到 180 度
  delay(1000);
}

Wire（I2C 通訊）

#include <Wire.h>

void setup() {
  Wire.begin();  // 初始化 I2C（作為主機）
}

void loop() {
  Wire.beginTransmission(0x27);  // 開始傳輸到地址 0x27
  Wire.write(0x00);              // 寫入資料
  Wire.endTransmission();        // 結束傳輸
}

除錯技巧

1. 使用 Serial 輸出除錯

void loop() {
  int value = analogRead(A0);

  // 印出變數值來除錯
  Serial.print("Sensor value: ");
  Serial.println(value);

  // 印出程式執行位置
  Serial.println(">>> 進入 if 判斷");

  if (value > 500) {
    Serial.println(">>> 條件成立");
    // ...
  }
}

2. LED 指示燈除錯

// 用 LED 表示程式狀態
digitalWrite(13, HIGH);  // 表示進入某段程式
// ... 執行程式 ...
digitalWrite(13, LOW);   // 表示離開

3. 條件編譯

#define DEBUG 1  // 設為 0 可關閉除錯訊息

  #if DEBUG
    Serial.println("除錯模式開啟");
  #endif
}

常見錯誤與解決

錯誤 原因 解決方法
expected ';' 少了分號 檢查上一行結尾
was not declared 變數未宣告 確認變數名稱和宣告位置
expected '}' 括號不配對 檢查大括號配對
invalid conversion 類型不符 檢查變數類型
redefinition 重複定義 移除重複的宣告
